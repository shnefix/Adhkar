import android.Manifest

fun timeLabel(ctx: Context, h: Int, m: Int): String =
    if (DateFormat.is24HourFormat(ctx)) "%02d:%02d".format(h, m)
    else "%d:%02d %s".format(if (h % 12 == 0) 12 else h % 12, m, if (h < 12) "AM" else "PM")

class RemindersFragment : Fragment(R.layout.fragment_reminders) {

    private val permission = registerForActivityResult(ActivityResultContracts.RequestPermission()) { granted ->
        if (!granted) Toast.makeText(requireContext(), "Notifications are off for Azkar in system settings.", Toast.LENGTH_SHORT).show()
    }

    private fun ensureNotificationPermission() {
        if (Build.VERSION.SDK_INT >= 33 &&
            requireContext().checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != android.content.pm.PackageManager.PERMISSION_GRANTED
        ) permission.launch(Manifest.permission.POST_NOTIFICATIONS)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val ctx = requireContext()
        bindNav(view, Tab.REMINDERS)

        view.findViewById<ToggleSwitch>(R.id.sw_all).apply {
            on = Store.notificationsOn()
            onToggle = { v ->
                Store.setNotificationsOn(v)
                if (v) ensureNotificationPermission()
                ReminderScheduler.rescheduleAll(ctx)   // cancels everything when off
                renderList()
            }
        }
        view.findViewById<View>(R.id.btn_add_rem).setOnClickListener { edit(null) }
        view.findViewById<View>(R.id.exact_row).setOnClickListener {
            if (Build.VERSION.SDK_INT >= 31) {
                startActivity(Intent(Settings.ACTION_REQUEST_SCHEDULE_EXACT_ALARM, Uri.parse("package:" + ctx.packageName)))
            }
        }

        // Show translation
        view.findViewById<ToggleSwitch>(R.id.sw_tr).apply {
            on = Store.showTranslation()
            onToggle = { Store.setShowTranslation(it) }
        }

        // Arabic text size
        val sizes = view.findViewById<LinearLayout>(R.id.sizes)
        val px = intArrayOf(20, 26, 32)
        val aria = arrayOf("Small Arabic text", "Medium Arabic text", "Large Arabic text")
        fun renderSizes() {
            sizes.removeAllViews()
            val z = Store.arabicSize()
            px.forEachIndexed { idx, size ->
                val on = idx == z
                val b = TextView(ctx, null, 0, R.style.ArB).apply {
                    text = "أب"
                    textSize = size.toFloat()
                    gravity = Gravity.CENTER
                    contentDescription = aria[idx]
                    setTextColor(ctx.getColor(if (on) R.color.bg else R.color.text))
                    val line = if (on) ctx.getColor(R.color.gold) else ctx.getColor(R.color.border2)
                    background = Ui.shape(ctx, if (on) ctx.getColor(R.color.gold) else null, line, 14f)
                    isClickable = true
                    isFocusable = true
                    setOnClickListener { Store.setArabicSize(idx); renderSizes() }
                }
                val p = LinearLayout.LayoutParams(0, Ui.pxi(ctx, 44f), 1f)
                if (idx > 0) p.marginStart = Ui.pxi(ctx, 8f)
                sizes.addView(b, p)
            }
        }
        renderSizes()
        renderList()
    }

    override fun onResume() {
        super.onResume()
        if (view != null) {
            ReminderScheduler.rescheduleAll(requireContext())   // picks up a newly granted exact-alarm permission
            renderList()
        }
    }

    private fun renderList() {
        val root = view ?: return
        val ctx = requireContext()
        val list = root.findViewById<LinearLayout>(R.id.reminders)
        list.removeAllViews()
        val all = Store.schedules()
        val globalOn = Store.notificationsOn()
        list.visibility = if (all.isEmpty()) View.GONE else View.VISIBLE
        root.findViewById<View>(R.id.rem_empty).visibility = if (all.isEmpty()) View.VISIBLE else View.GONE
        root.findViewById<TextView>(R.id.notif_sub).text =
            if (!globalOn) "All reminders are paused" else "${all.count { it.enabled }} of ${all.size} reminders on"

        all.forEachIndexed { idx, s ->
            val v = layoutInflater.inflate(R.layout.item_reminder, list, false)
            v.findViewById<View>(R.id.divider).setBackgroundColor(if (idx == 0) 0 else ctx.getColor(R.color.border))
            v.findViewById<TextView>(R.id.title).text = s.title
            v.findViewById<TextView>(R.id.time).text = timeLabel(ctx, s.hour, s.minute) + " · Every day"
            v.findViewById<View>(R.id.row).apply {
                alpha = if (globalOn) 1f else 0.5f
                setOnClickListener { edit(s) }
            }
            v.findViewById<ToggleSwitch>(R.id.sw).apply {
                contentDescription = s.title + " reminder"
                on = s.enabled
                onToggle = { isOn ->
                    val updated = s.copy(enabled = isOn)
                    Store.upsertSchedule(updated)
                    if (isOn) ensureNotificationPermission()
                    ReminderScheduler.apply(ctx, updated)
                    renderList()
                }
            }
            list.addView(v)
        }

        val needExact = Build.VERSION.SDK_INT >= 31 &&
            !(ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager).canScheduleExactAlarms()
        root.findViewById<View>(R.id.exact_row).visibility =
            if (needExact && globalOn && all.any { it.enabled }) View.VISIBLE else View.GONE
    }

    private fun edit(existing: Schedule?) {
        val ctx = requireContext()
        val v = layoutInflater.inflate(R.layout.dialog_reminder, null)
        val title = v.findViewById<EditText>(R.id.r_title)
        val msg = v.findViewById<EditText>(R.id.r_msg)
        val timeTv = v.findViewById<TextView>(R.id.r_time)
        var h = existing?.hour ?: 8
        var m = existing?.minute ?: 0
        title.setText(existing?.title ?: "")
        msg.setText(existing?.message ?: "")
        fun showTime() { timeTv.text = timeLabel(ctx, h, m) }
        showTime()
        timeTv.setOnClickListener {
            TimePickerDialog(ctx, { _, hh, mm -> h = hh; m = mm; showTime() }, h, m, DateFormat.is24HourFormat(ctx)).show()
        }

        val b = AlertDialog.Builder(ctx)
            .setTitle(if (existing == null) "New reminder" else "Edit reminder")
            .setView(v)
            .setPositiveButton("Save", null)
            .setNegativeButton("Cancel", null)
        if (existing != null) b.setNeutralButton("Delete") { _, _ ->
            ReminderScheduler.cancel(ctx, existing.id)   // cancel the alarm before forgetting it
            Store.deleteSchedule(existing.id)
            renderList()
        }
        val dlg = b.create()
        dlg.setOnShowListener {
            dlg.getButton(DialogInterface.BUTTON_POSITIVE).setOnClickListener {
                val t = title.text.toString().trim()
                if (t.isEmpty()) { title.error = "Enter a title"; return@setOnClickListener }
                val text = msg.text.toString().trim().ifEmpty { "Time for your azkar · tap to read" }
                val s = (existing ?: Schedule(Store.newScheduleId(), t, text, h, m, true, null))
                    .copy(title = t, message = text, hour = h, minute = m)
                Store.upsertSchedule(s)
                ReminderScheduler.cancel(ctx, s.id)   // drop the old alarm, then set the new time
                ReminderScheduler.apply(ctx, s)
                if (s.enabled) ensureNotificationPermission()
                renderList()
                dlg.dismiss()
            }
        }
        dlg.show()
    }
}

package com.azkar.app.data

import android.content.Context
import android.content.SharedPreferences
import org.json.JSONArray
import org.json.JSONObject
import java.util.UUID

/** Persistent local storage (SharedPreferences + JSON). Survives app restarts. */
object Store {
    private lateinit var sp: SharedPreferences

    private const val K_ZIKRS = "zikrs"
    private const val K_FOLDERS = "folders"
    private const val K_ORDER = "home_order"
    private const val K_HIDDEN = "home_hidden"
    private const val K_PLACE = "home_place"
    private const val K_TRANS = "show_translation"
    private const val K_SIZE = "arabic_size"
    private const val K_REM = "reminder_"
    private const val K_NOTIF = "notif_on"
    private const val K_SCHED = "schedules"
    private const val K_MIG = "builtins_v2"

    fun init(context: Context) {
        if (::sp.isInitialized) return
        sp = context.applicationContext.getSharedPreferences("azkar", Context.MODE_PRIVATE)
        if (!sp.contains(K_ZIKRS)) seed()
        migrateBuiltins()
        migrateSchedules()
    }

    private fun seed() {
        saveFolders(listOf(Folder("family", "Family"), Folder("friday", "Friday")))
        saveZikrs(
            listOf(
                Zikr("seed1", "أَسْتَغْفِرُ اللَّهَ", "I seek forgiveness from Allah.", 100, null),
                Zikr("seed2", "لَا حَوْلَ وَلَا قُوَّةَ إِلَّا بِاللَّهِ", "There is no power and no strength except with Allah.", 10, "family"),
                Zikr("seed3", "اللَّهُمَّ صَلِّ وَسَلِّمْ عَلَى نَبِيِّنَا مُحَمَّدٍ", "O Allah, send prayers and peace upon our Prophet Muhammad.", 10, "friday")
            )
        )
    }

    /** One time: the built-in collections become ordinary, fully editable folders with the same ids. */
    private fun migrateBuiltins() {
        if (sp.getBoolean(K_MIG, false)) return
        val have = folders()
        val missing = Builtins.library.filter { c -> have.none { it.id == c.key } }
        saveFolders(missing.map { Folder(it.key, it.title) } + have)
        val old = zikrs()
        val seeded = missing.flatMap { c ->
            c.items.mapIndexed { n, r -> Zikr("b_${c.key}_$n", r.arabic, "", r.target, c.key) }
        }
        saveZikrs(seeded + old)
        setHidden(hidden() + setOf("morning", "worry", "forgiveness"))   // they were not on Home before
        sp.edit().putBoolean(K_MIG, true).apply()
    }

    /** One time: the three fixed reminders become editable schedules (same on/off state). */
    private fun migrateSchedules() {
        if (sp.contains(K_SCHED)) return
        saveSchedules(
            listOf(
                Schedule("m", "Morning azkar", "Time for your morning azkar · tap to read", 6, 0, reminderOn("m", true), "morning"),
                Schedule("e", "Evening azkar", "Time for your evening azkar · tap to read", 17, 0, reminderOn("e", true), "evening"),
                Schedule("s", "Before sleep", "Time for your before sleep azkar · tap to read", 22, 0, reminderOn("s", false), "sleep")
            )
        )
    }

    // ---------- Azkar ----------
    fun zikrs(): List<Zikr> {
        val arr = JSONArray(sp.getString(K_ZIKRS, "[]"))
        return (0 until arr.length()).map {
            val o = arr.getJSONObject(it)
            Zikr(
                o.getString("id"), o.getString("ar"), o.optString("note", ""), o.optInt("n", 1),
                if (o.isNull("f")) null else o.getString("f")
            )
        }
    }

    private fun saveZikrs(list: List<Zikr>) {
        val arr = JSONArray()
        list.forEach {
            arr.put(JSONObject().put("id", it.id).put("ar", it.arabic).put("note", it.note).put("n", it.repeat)
                .put("f", it.folderId ?: JSONObject.NULL))
        }
        sp.edit().putString(K_ZIKRS, arr.toString()).apply()
    }

    fun zikr(id: String): Zikr? = zikrs().firstOrNull { it.id == id }
    fun zikrsIn(folderId: String): List<Zikr> = zikrs().filter { it.folderId == folderId }

    fun addZikr(arabic: String, note: String, repeat: Int, folderId: String?): Zikr {
        zikrs().firstOrNull { it.arabic == arabic && it.folderId == folderId }?.let {
            val merged = it.copy(note = note.ifBlank { it.note }, repeat = repeat)
            updateZikr(merged)
            return merged
        }
        val z = Zikr(UUID.randomUUID().toString(), arabic, note, repeat, folderId)
        saveZikrs(zikrs() + z)
        return z
    }

    fun updateZikr(z: Zikr) = saveZikrs(zikrs().map { if (it.id == z.id) z else it })
    fun deleteZikr(id: String) = saveZikrs(zikrs().filter { it.id != id })

    /** Takes a zikr out of its folder (it stays saved as "Anytime"). */
    fun unfile(id: String) = saveZikrs(zikrs().map { if (it.id == id) it.copy(folderId = null) else it })

    /** Reorders the zikrs of one folder; other azkar keep their relative position. */
    fun reorderFolder(folderId: String, orderedIds: List<String>) {
        val all = zikrs()
        val byId = all.associateBy { it.id }
        val queue = orderedIds.mapNotNull { byId[it] }.toMutableList()
        saveZikrs(all.map { if (it.folderId == folderId && queue.isNotEmpty()) queue.removeAt(0) else it })
    }

    // ---------- Folders ----------
    fun folders(): List<Folder> {
        val arr = JSONArray(sp.getString(K_FOLDERS, "[]"))
        return (0 until arr.length()).map { Folder(arr.getJSONObject(it).getString("id"), arr.getJSONObject(it).getString("name")) }
    }

    private fun saveFolders(list: List<Folder>) {
        val arr = JSONArray()
        list.forEach { arr.put(JSONObject().put("id", it.id).put("name", it.name)) }
        sp.edit().putString(K_FOLDERS, arr.toString()).apply()
    }

    fun folder(id: String): Folder? = folders().firstOrNull { it.id == id }

    fun addFolder(name: String): Folder {
        val f = Folder(UUID.randomUUID().toString(), name)
        saveFolders(folders() + f)
        return f
    }

    fun renameFolder(id: String, name: String) = saveFolders(folders().map { if (it.id == id) it.copy(name = name) else it })

    /** Moves a folder up (-1) or down (+1) in the Library order. */
    fun moveFolder(id: String, d: Int) {
        val l = folders().toMutableList()
        val i = l.indexOfFirst { it.id == id }
        val j = i + d

    fun importFolders(plan: List<ImportJson.FolderIn>): ImportSummary {
        var created = 0; var merged = 0; var added = 0; var skipped = 0
        val allZ = zikrs().toMutableList()
        val fl = folders().toMutableList()
        plan.forEach { f ->
            val name = f.name.trim()
            val existing = fl.firstOrNull { it.name.trim().equals(name, ignoreCase = true) }
            val folder = existing ?: Folder(UUID.randomUUID().toString(), name).also { fl.add(it); created++ }

            f.items.forEach { item ->
                if (allZ.any { z -> z.folderId == folder.id && z.arabic == item.text }) skipped++
                else { allZ.add(Zikr(UUID.randomUUID().toString(), item.text, "", item.count, folder.id)); added++ }

object ReminderScheduler {
    const val CHANNEL = "azkar_reminders_v2"
    private const val ACTION = "com.azkar.app.REMINDER"

    fun createChannel(ctx: Context) {
        val ch = NotificationChannel(CHANNEL, "Daily reminders", NotificationManager.IMPORTANCE_HIGH).apply {
            description = "Your azkar reminders"
            enableVibration(true)
            setShowBadge(true)
        }
        ctx.getSystemService(NotificationManager::class.java).createNotificationChannel(ch)
    }

    private fun alarms(ctx: Context) = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    /** Exact alarms need a user grant on Android 12+. */
    fun canExact(ctx: Context) = Build.VERSION.SDK_INT < 31 || alarms(ctx).canScheduleExactAlarms()

    // One alarm per schedule id (unique data URI), so editing replaces and deleting cancels exactly that alarm.
    private fun intentFor(ctx: Context, id: String) =
        Intent(ctx, ReminderReceiver::class.java).setAction(ACTION).setData(Uri.parse("azkar://reminder/$id")).putExtra("id", id)

    private fun pending(ctx: Context, id: String): PendingIntent =
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)

    /** Alarms from the old fixed reminders (request codes 0..2, no action). */
    private fun cancelLegacy(ctx: Context) {
        for (i in 0..2) {
            PendingIntent.getBroadcast(ctx, i, Intent(ctx, ReminderReceiver::class.java), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
                ?.let { alarms(ctx).cancel(it); it.cancel() }
        }
    }

    fun rescheduleAll(ctx: Context) {
        cancelLegacy(ctx)
        Store.schedules().forEach { apply(ctx, it) }
    }

    /** Schedules or cancels one reminder according to the global and individual switches. */
    fun apply(ctx: Context, s: Schedule) {
        if (Store.notificationsOn() && s.enabled) schedule(ctx, s) else cancel(ctx, s.id)
    }

    fun schedule(ctx: Context, s: Schedule) {
        val cal = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, s.hour); set(Calendar.MINUTE, s.minute)
            set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0)
            if (timeInMillis <= System.currentTimeMillis()) add(Calendar.DAY_OF_YEAR, 1)
        }
        val am = alarms(ctx)
        if (canExact(ctx)) am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, cal.timeInMillis, pending(ctx, s.id))
        else am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, cal.timeInMillis, pending(ctx, s.id))
    }

    fun cancel(ctx: Context, id: String) {
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
            ?.let { alarms(ctx).cancel(it); it.cancel() }
    }

    @SuppressLint("MissingPermission")
    fun notify(ctx: Context, s: Schedule) {
        if (Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(ctx, android.Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) return
        val nid = 100 + (s.id.hashCode() and 0xFFFF)
        val open = PendingIntent.getActivity(
            ctx, nid,
            Intent(ctx, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
                .apply { s.open?.let { putExtra("open", it) } },
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val ar = s.open?.let { Builtins.byKey(it)?.homeArabic } ?: ""
        val rv = RemoteViews(ctx.packageName, R.layout.notification_reminder).apply {
            setTextViewText(R.id.n_title, s.title)
            setTextViewText(R.id.n_body, s.message)
            setTextViewText(R.id.n_ar, ar)
            setViewVisibility(R.id.n_ar, if (ar.isEmpty()) View.GONE else View.VISIBLE)
        }
        val n = NotificationCompat.Builder(ctx, CHANNEL)
            .setSmallIcon(R.drawable.ic_bell_1p75)
            .setColor(ctx.getColor(R.color.gold))
            .setContentTitle(s.title)
            .setContentText(s.message)
            .setStyle(NotificationCompat.DecoratedCustomViewStyle())
            .setCustomContentView(rv)
            .setCustomBigContentView(rv)
            .setCategory(NotificationCompat.CATEGORY_REMINDER)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
            .setContentIntent(open)
            .setAutoCancel(true)
            .build()
        NotificationManagerCompat.from(ctx).notify(nid, n)
    }
}

class ReminderReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        Store.init(ctx)
        ReminderScheduler.createChannel(ctx)
        val id = intent.getStringExtra("id") ?: return          // ignores old-style alarms
        val s = Store.schedule(id) ?: return                    // deleted schedule: nothing to show
        if (!Store.notificationsOn() || !s.enabled) return
        ReminderScheduler.notify(ctx, s)
        ReminderScheduler.schedule(ctx, s)                      // next day
    }
}

class BootReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        if (intent.action == Intent.ACTION_BOOT_COMPLETED) {
            Store.init(ctx)
            ReminderScheduler.createChannel(ctx)
            ReminderScheduler.rescheduleAll(ctx)
        }
    }
}

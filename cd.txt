        val cases = listOf(
            WidgetStyle.CLASSIC to WidgetKind.SMALL,
            WidgetStyle.CLASSIC to WidgetKind.MEDIUM,
            WidgetStyle.CLASSIC to WidgetKind.LARGE,
            WidgetStyle.CHECKLIST to WidgetKind.MEDIUM,
            WidgetStyle.CHECKLIST to WidgetKind.LARGE,
            WidgetStyle.TASBIH to WidgetKind.SMALL,
            WidgetStyle.TASBIH to WidgetKind.LARGE,
            WidgetStyle.CALLIGRAPHY to WidgetKind.SMALL,
            WidgetStyle.CALLIGRAPHY to WidgetKind.MEDIUM,
        )

        for (theme in intArrayOf(ThemeMode.CURRENT, ThemeMode.OLED)) {
            Store.setTheme(theme)
            for ((style, kind) in cases) {
                val (width, height) = when (kind) {
                    WidgetKind.SMALL -> 158f to 158f
                    WidgetKind.MEDIUM -> 338f to 158f
                    WidgetKind.LARGE -> 338f to 354f
                }
                val bitmap = WidgetRenderer(context).render(style, kind, model, width, height, 1f)
                assertEquals(width.toInt(), bitmap.width)
                assertEquals(height.toInt(), bitmap.height)
                assertTrue(bitmap.hasAlpha())
                assertTrue(bitmap.getPixel(1, bitmap.height / 2) != 0)
                bitmap.recycle()
            }
        }
    }

    @Test
    fun themeColorsMeetReadableContrast() {
        val resources = context.resources
        val palettes = listOf(
            R.color.widget_teal_background to R.color.widget_teal_text,
            R.color.widget_teal_background to R.color.widget_teal_muted,
            R.color.widget_oled_background to R.color.widget_oled_text,
            R.color.widget_oled_background to R.color.widget_oled_muted,
        )
        for ((backgroundId, foregroundId) in palettes) {
            val background = resources.getColor(backgroundId, context.theme)
            val foreground = resources.getColor(foregroundId, context.theme)
            assertTrue(contrast(background, foreground) >= 4.5)
        }

    }

    private fun contrast(first: Int, second: Int): Double {
        fun channel(value: Int): Double {
            val normalized = value / 255.0
            return if (normalized <= 0.03928) normalized / 12.92 else Math.pow((normalized + 0.055) / 1.055, 2.4)
        }
        fun luminance(color: Int): Double =
            0.2126 * channel(color shr 16 and 0xFF) +
                0.7152 * channel(color shr 8 and 0xFF) +
                0.0722 * channel(color and 0xFF)
        val lighter = maxOf(luminance(first), luminance(second))
        val darker = minOf(luminance(first), luminance(second))
        return (lighter + 0.05) / (darker + 0.05)
    }

package com.azkar.app

import android.Manifest
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Build
import android.os.Bundle
import android.view.WindowManager
import android.widget.Toast
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.WindowCompat
import androidx.fragment.app.Fragment
import androidx.fragment.app.FragmentManager
import androidx.lifecycle.lifecycleScope
import androidx.lifecycle.repeatOnLifecycle
import kotlinx.coroutines.launch
import kotlinx.coroutines.Dispatchers
import com.azkar.app.data.Store
import com.azkar.app.ui.HomeFragment
import com.azkar.app.ui.LibraryFragment
import com.azkar.app.ui.MyAzkarFragment
import com.azkar.app.ui.ReaderFragment
import com.azkar.app.ui.Tab
import com.azkar.app.widgets.WidgetUpdater
import dagger.hilt.android.AndroidEntryPoint

@AndroidEntryPoint
class MainActivity : AppCompatActivity() {
    private var currentTab: Tab = Tab.HOME
    private val notifPermission = registerForActivityResult(ActivityResultContracts.RequestPermission()) { }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        currentTab = savedInstanceState?.getInt("tab")?.let { Tab.values().getOrNull(it) } ?: Tab.HOME
        Store.init(this)
        if (Build.VERSION.SDK_INT >= 28) {
            window.attributes = window.attributes.apply {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
        }
        if (Build.VERSION.SDK_INT >= 33 &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) notifPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
        WindowCompat.setDecorFitsSystemWindows(window, false)
        WindowCompat.getInsetsController(window, window.decorView).apply {
            isAppearanceLightStatusBars = false
            isAppearanceLightNavigationBars = false
        }
        var contentShown = false
        lifecycleScope.launch {
            try {
                repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
                    if (contentShown) return@repeatOnLifecycle
                    Store.awaitReady()
                    ThemeMode.apply(Store.theme())
                    if (Store.consumeRecoveryNotice()) {
                        Toast.makeText(this@MainActivity, R.string.data_recovered, Toast.LENGTH_LONG).show()
                    }
                    setContentView(R.layout.activity_main)
                    if (savedInstanceState == null) {
                        showRoot(HomeFragment())
                        handleIntent(intent)
                    }
                    contentShown = true
                }
            } catch (e: Exception) {
                android.util.Log.e("Azkar", "Unable to initialize application data", e)
                Toast.makeText(this@MainActivity, "Azkar data could not be loaded.", Toast.LENGTH_LONG).show()
            }
        }
    }

    override fun onSaveInstanceState(outState: Bundle) {
        super.onSaveInstanceState(outState)
        outState.putInt("tab", currentTab.ordinal)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        lifecycleScope.launch {
            try {
                repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
                    Store.awaitReady()
                    handleIntent(intent)
                    return@repeatOnLifecycle
                }
            } catch (e: Exception) {
                android.util.Log.e("Azkar", "Unable to handle application deep link", e)
            }
        }
    }

    private fun handleIntent(i: Intent?) {
        val key = i?.getStringExtra("open") ?: return
        i.removeExtra("open")
        showRoot(HomeFragment())
        if (Store.readerItems(key).isNotEmpty()) push(ReaderFragment.of(key))   // deleted or empty folder: just stay on Home
    }

    private fun showRoot(f: Fragment) {
        supportFragmentManager.popBackStackImmediate(null, FragmentManager.POP_BACK_STACK_INCLUSIVE)
        supportFragmentManager.beginTransaction().replace(R.id.container, f).commit()
    }

    fun openTab(tab: Tab) {
        if (tab == currentTab && supportFragmentManager.backStackEntryCount == 0) return
        currentTab = tab
        showRoot(
            when (tab) {
                Tab.HOME -> HomeFragment()
                Tab.LIBRARY -> LibraryFragment()
                Tab.MY -> MyAzkarFragment()
            }
        )
    }

    fun push(f: Fragment) {
        supportFragmentManager.beginTransaction().replace(R.id.container, f).addToBackStack(null).commit()
    }

    fun pop() {
        android.util.Log.d("AzkarNav", "pop() called from:", Throwable())
        supportFragmentManager.popBackStack()
    }

    /** The last zikr of a collection was finished: back to Home. */
    fun finishReading() {
        currentTab = Tab.HOME
        showRoot(HomeFragment())
    }
    /** Resets the back stack and shows Home. */
    fun goHome() {
        android.util.Log.d("AzkarNav", "goHome() called from:", Throwable())
        currentTab = Tab.HOME
        showRoot(HomeFragment())
    }

    override fun onStop() {
        super.onStop()
        lifecycleScope.launch(Dispatchers.IO) {
            runCatching {
                Store.awaitReady()
                WidgetUpdater.updateAll(this@MainActivity)
            }.onFailure { android.util.Log.e("Azkar", "Widget refresh failed", it) }
        }
    }
}

package com.azkar.app.reminders

import android.annotation.SuppressLint
import android.app.AlarmManager
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.net.Uri
import android.os.Build
import android.view.View
import android.widget.RemoteViews
import androidx.core.app.NotificationCompat
import androidx.core.app.NotificationManagerCompat
import androidx.core.content.ContextCompat
import com.azkar.app.MainActivity
import com.azkar.app.R
import com.azkar.app.ThemeMode
import com.azkar.app.data.Builtins
import com.azkar.app.data.Schedule
import com.azkar.app.data.Store
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch

object ReminderScheduler {
    const val CHANNEL = "azkar_reminders_v2"
    private const val ACTION = "com.azkar.app.REMINDER"

    fun createChannel(ctx: Context) {
        val ch = NotificationChannel(CHANNEL, "Daily reminders", NotificationManager.IMPORTANCE_HIGH).apply {
            description = "Your azkar reminders"
            enableVibration(true)
            setShowBadge(true)
        }
        ctx.getSystemService(NotificationManager::class.java).createNotificationChannel(ch)
    }

    private fun alarms(ctx: Context) = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    /** Exact alarms need a user grant on Android 12+. */
    fun canExact(ctx: Context) = Build.VERSION.SDK_INT < 31 || alarms(ctx).canScheduleExactAlarms()

    // One alarm per schedule id (unique data URI), so editing replaces and deleting cancels exactly that alarm.
    private fun intentFor(ctx: Context, id: String) =
        Intent(ctx, ReminderReceiver::class.java).setAction(ACTION).setData(Uri.parse("azkar://reminder/$id")).putExtra("id", id)

    private fun pending(ctx: Context, id: String): PendingIntent =
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)

    /** Alarms from the old fixed reminders (request codes 0..2, no action). */
    private fun cancelLegacy(ctx: Context) {
        for (i in 0..2) {
            PendingIntent.getBroadcast(ctx, i, Intent(ctx, ReminderReceiver::class.java), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
                ?.let { alarms(ctx).cancel(it); it.cancel() }
        }
    }

    fun rescheduleAll(ctx: Context) {
        cancelLegacy(ctx)
        Store.schedules().forEach { apply(ctx, it) }
    }

    /** Schedules or cancels one reminder according to the global and individual switches. */
    fun apply(ctx: Context, s: Schedule) {
        if (Store.notificationsOn() && s.enabled) schedule(ctx, s) else cancel(ctx, s.id)
    }

    fun schedule(ctx: Context, s: Schedule) {
        val am = alarms(ctx)
        val at = ReminderTime.next(s.hour, s.minute)
        if (canExact(ctx)) am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pending(ctx, s.id))
        else am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pending(ctx, s.id))
    }

    fun cancel(ctx: Context, id: String) {
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
            ?.let { alarms(ctx).cancel(it); it.cancel() }
    }

    @SuppressLint("MissingPermission")
    fun notify(ctx: Context, s: Schedule) {
        if (Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(ctx, android.Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) return
        val nid = 100 + (s.id.hashCode() and 0xFFFF)
        val open = PendingIntent.getActivity(
            ctx, nid,
            Intent(ctx, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
                .apply { s.open?.let { putExtra("open", it) } },
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val ar = s.open?.let { Builtins.byKey(it)?.homeArabic } ?: ""
        // The system UI inflates RemoteViews with the device's night setting, not ours,
        // so each theme has its own layout built only from colours that never flip.
        val layout = if (Store.theme() == ThemeMode.OLED) R.layout.notification_reminder_oled else R.layout.notification_reminder
        val rv = RemoteViews(ctx.packageName, layout).apply {
            setTextViewText(R.id.n_title, s.title)
            setTextViewText(R.id.n_body, s.message)
            setTextViewText(R.id.n_ar, ar)
            setViewVisibility(R.id.n_ar, if (ar.isEmpty()) View.GONE else View.VISIBLE)
        }
        val n = NotificationCompat.Builder(ctx, CHANNEL)
            .setSmallIcon(R.drawable.ic_bell_1p75)
            .setColor(ctx.getColor(R.color.gold))
            .setContentTitle(s.title)
            .setContentText(s.message)
            .setCustomContentView(rv)
            .setCustomBigContentView(rv)
            .setCategory(NotificationCompat.CATEGORY_REMINDER)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
            .setContentIntent(open)
            .setAutoCancel(true)
            .build()
        NotificationManagerCompat.from(ctx).notify(nid, n)
    }
}

class ReminderReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        Store.init(ctx)
        val pending = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Store.awaitReady()
                ReminderScheduler.createChannel(ctx)
                val id = intent.getStringExtra("id") ?: return@launch
                val s = Store.schedule(id) ?: return@launch
                if (!Store.notificationsOn() || !s.enabled) return@launch
                ReminderScheduler.notify(ctx, s)
                ReminderScheduler.schedule(ctx, s)
            } finally {
                pending.finish()
            }
        }
    }
}

class BootReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        if (intent.action == Intent.ACTION_BOOT_COMPLETED) {
            Store.init(ctx)
            val pending = goAsync()
            CoroutineScope(Dispatchers.IO).launch {
                try {
                    Store.awaitReady()
                    ReminderScheduler.createChannel(ctx)
                    ReminderScheduler.rescheduleAll(ctx)
                } finally {
                    pending.finish()
                }
            }
        }
    }
}

package com.azkar.app.reminders

import java.util.Calendar

object ReminderTime {
    fun next(hour: Int, minute: Int, now: Long = System.currentTimeMillis()): Long {
        val cal = Calendar.getInstance().apply {
            timeInMillis = now
            set(Calendar.HOUR_OF_DAY, hour)
            set(Calendar.MINUTE, minute)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
            if (timeInMillis <= now) add(Calendar.DAY_OF_YEAR, 1)
        }
        return cal.timeInMillis
    }
}

/** 12dp gaps between cards in a 2-column grid. */
class GridGap(private val dp: Int, private val span: Int = 2) : RecyclerView.ItemDecoration() {
    override fun getItemOffsets(out: Rect, v: View, parent: RecyclerView, s: RecyclerView.State) {
        val pos = parent.getChildAdapterPosition(v)
        if (pos == RecyclerView.NO_POSITION) return
        val px = (dp * v.resources.displayMetrics.density).toInt()
        val col = pos % span
        out.left = if (col == 0) 0 else px / 2
        out.right = if (col == 0) px / 2 else 0
        out.top = if (pos >= span) px else 0
    }
}

package com.azkar.app.ui

/** Pure navigation rules of the reader (no Android classes), so they can be unit tested. */
class ReaderNav(private val targets: List<Int>, start: Int = 0, count: Int = 0) {
    enum class Outcome { MOVED, FINISH, BOUNCE }

    var index: Int = start.coerceIn(0, (targets.size - 1).coerceAtLeast(0)); private set
    var count: Int = count; private set

    val last get() = targets.size - 1
    val isLast get() = index == last
    /** The repetition count of the CURRENT zikr is complete. */
    val isDone get() = targets[index] <= 1 || count >= targets[index]

    /** Counter tap. Returns true if the count changed. */
    fun tapRing(): Boolean { if (isDone) return false; count++; return true }

    fun canPage(dir: Int) = if (dir > 0) index < last else index > 0
    /** Leaving the reader needs BOTH: last zikr of the collection AND its own count complete. */
    fun canFinish() = isLast && isDone

    /** Tap on the card or swipe left. */
    fun next(): Outcome = when {
        canPage(1) -> { index++; count = 0; Outcome.MOVED }
        canFinish() -> Outcome.FINISH
        else -> Outcome.BOUNCE
    }

    /** Swipe right. Never finishes. */
    fun prev(): Outcome =
        if (canPage(-1)) { index--; count = 0; Outcome.MOVED } else Outcome.BOUNCE
}

    fun px(ctx: Context, dp: Float): Float = dp * ctx.resources.displayMetrics.density
    fun pxi(ctx: Context, dp: Float): Int = px(ctx, dp).toInt()
    fun color(ctx: Context, @ColorRes id: Int) = ctx.getColor(id)

    fun shape(ctx: Context, fill: Int?, stroke: Int?, radiusDp: Float, dashed: Boolean = false): GradientDrawable =
        GradientDrawable().apply {

            cornerRadius = px(ctx, radiusDp)

            if (stroke != null) {
                if (dashed) setStroke(pxi(ctx, 1f), stroke, px(ctx, 3f), px(ctx, 3f))
                else setStroke(pxi(ctx, 1f), stroke)
            }
        }

        LinearLayout.LayoutParams(w, h).also { if (ctx != null && top != 0) it.topMargin = pxi(ctx, top.toFloat()) }

    /** Rows of [cols] equally wide cells with [gap]dp between them. */
    fun grid(ctx: Context, cells: List<View>, cols: Int, gap: Int): LinearLayout {
        val root = LinearLayout(ctx).apply { orientation = LinearLayout.VERTICAL }
        cells.chunked(cols).forEachIndexed { r, rowCells ->
            val row = LinearLayout(ctx).apply { orientation = LinearLayout.HORIZONTAL }
            for (c in 0 until cols) {
                val cell = rowCells.getOrNull(c) ?: View(ctx)
                val p = LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f)
                if (c > 0) p.marginStart = pxi(ctx, gap.toFloat())
                row.addView(cell, p)
            }
            root.addView(row, LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.WRAP_CONTENT).also {
                if (r > 0) it.topMargin = pxi(ctx, gap.toFloat())
            })
        }
        return root
    }

    fun chip(ctx: Context, label: String, on: Boolean, isNew: Boolean, onClick: () -> Unit): TextView {

            img.setImageResource(vector)

    }

/** Card-follows-finger swipe layout. dir: +1 = next (swipe left), -1 = previous (swipe right). */
class SwipeLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : LinearLayout(ctx, attrs) {
    var card: View? = null
    var canPage: ((Int) -> Boolean)? = null
    var onPage: ((Int) -> Unit)? = null
    var tapEnabled = true
    var canFinish: (() -> Boolean)? = null
    var onFinish: (() -> Unit)? = null
    private var rawDx = 0f
    private var finishing = false

    private val vc = android.view.ViewConfiguration.get(ctx)
    private val slop = vc.scaledTouchSlop
    private val minFling = vc.scaledMinimumFlingVelocity * 8f
    private val dp = resources.displayMetrics.density
    private var downX = 0f
    private var downY = 0f
    private var dragging = false
    private var tapOk = false
    private var busy = false
    private var vt: android.view.VelocityTracker? = null

    /** Stops the system back gesture from stealing swipes that start at the screen edge. */
    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        super.onLayout(changed, l, t, r, b)
        if (android.os.Build.VERSION.SDK_INT >= 29) {
            val w = r - l
            val h = b - t
            val c = card
            val cy = if (c != null) (c.top + c.bottom) / 2 else h / 2
            val half = (100 * dp).toInt()
            val top = (cy - half).coerceAtLeast(0)
            val bottom = (cy + half).coerceAtMost(h)
            val edge = (40 * dp).toInt()
            setSystemGestureExclusionRects(
                listOf(android.graphics.Rect(0, top, edge, bottom), android.graphics.Rect(w - edge, top, w, bottom))
            )
        }
    }

    private fun inCard(x: Float, y: Float): Boolean {
        val c = card ?: return false
        return x >= c.left && x <= c.right && y >= c.top && y <= c.bottom
    }

    private fun far(e: MotionEvent) =
        kotlin.math.hypot(e.x - downX, e.y - downY) > slop

    private fun horizontal(e: MotionEvent) =
        kotlin.math.abs(e.x - downX) > slop && kotlin.math.abs(e.x - downX) > kotlin.math.abs(e.y - downY) * 1.2f

    private fun start(e: MotionEvent) {
        downX = e.x; downY = e.y; dragging = false
        tapOk = inCard(e.x, e.y)
        vt?.recycle(); vt = android.view.VelocityTracker.obtain(); vt?.addMovement(e)
    }

    private fun finishTap(e: MotionEvent) {
        if (tapEnabled && tapOk && e.eventTime - e.downTime < android.view.ViewConfiguration.getLongPressTimeout()) next()
        tapOk = false
    }

    private fun cleanup() { vt?.recycle(); vt = null }

    override fun onInterceptTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
            }
            MotionEvent.ACTION_UP -> { if (!dragging) finishTap(e); cleanup() }
            MotionEvent.ACTION_CANCEL -> { tapOk = false; cleanup() }
        }
        return dragging
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        if (vt == null) vt = android.view.VelocityTracker.obtain()
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
                if (dragging) drag(e.x - downX)
            }
            MotionEvent.ACTION_UP -> {
                vt?.addMovement(e)
                if (dragging) release(true) else finishTap(e)
                dragging = false; cleanup()
            }
            MotionEvent.ACTION_CANCEL -> {
                if (dragging) release(false)
                dragging = false; tapOk = false; cleanup()
            }
        }
        return true
    }

    private fun dirOf(t: Float) = if (t < 0f) 1 else -1   // finger moved left -> next

    private fun drag(dx: Float) {
        val c = card ?: return
        rawDx = dx
        val d = dx - kotlin.math.sign(dx) * slop
        val ok = canPage?.invoke(dirOf(d)) ?: false
        val t = if (ok) d else d * 0.25f          // rubber band at the first/last azkar
        c.translationX = t
        c.alpha = 1f - 0.6f * minOf(kotlin.math.abs(t) / width.coerceAtLeast(1), 1f)
    }

    private fun release(up: Boolean) {
        val c = card ?: return
        vt?.computeCurrentVelocity(1000)
        val vx = vt?.xVelocity ?: 0f
        val t = c.translationX
        val dir = dirOf(t)
        val ok = canPage?.invoke(dir) ?: false
        val farEnough = kotlin.math.abs(t) > width * 0.28f
        val flung = kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(t) && kotlin.math.abs(t) > slop * 2
        val rawFar = kotlin.math.abs(rawDx) > 48 * dp ||
            (kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(rawDx))
        val finish = up && dir == 1 && !ok && rawFar && (canFinish?.invoke() == true)
        rawDx = 0f
        if (up && ok && (farEnough || flung)) commit(c, dir)
        else if (finish) { c.animate().cancel(); c.translationX = 0f; c.alpha = 1f; fireFinish() }
        else c.animate().translationX(0f).alpha(1f).setDuration(200)
            .setInterpolator(android.view.animation.DecelerateInterpolator()).start()
    }

    private fun fireFinish() {
        if (finishing) return
        finishing = true
        post { onFinish?.invoke() }   // never change fragments in the middle of touch dispatch
    }

    /** Next zikr; on the last one, leaves only if its count is complete, otherwise a small bounce. */
    fun next() {
        val c = card ?: return
        if (busy || finishing) return
        if (canPage?.invoke(1) == true) commit(c, 1)
        else if (canFinish?.invoke() == true) fireFinish()
        else c.animate().translationX(-12 * dp).setDuration(90)
            .withEndAction { c.animate().translationX(0f).setDuration(160).start() }.start()
    }

    private fun commit(c: View, dir: Int) {
        busy = true
        val w = width.toFloat()
        c.animate().translationX(-dir * w).alpha(0f).setDuration(140)
            .setInterpolator(android.view.animation.AccelerateInterpolator())
            .withEndAction {
                onPage?.invoke(dir)
                c.translationX = dir * w * 0.35f   // new card enters from the side you swiped toward
                c.alpha = 0f
                c.animate().translationX(0f).alpha(1f).setDuration(220)
                    .setInterpolator(android.view.animation.DecelerateInterpolator())
                    .withEndAction { busy = false }.start()
            }.start()
    }
}

/** Wrapping row layout with 8dp gaps (flex-wrap). */
class FlowLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : ViewGroup(ctx, attrs) {
    private val gap = (8 * resources.displayMetrics.density).toInt()

    override fun onMeasure(wSpec: Int, hSpec: Int) {
        val maxW = MeasureSpec.getSize(wSpec) - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            c.measure(MeasureSpec.makeMeasureSpec(maxW, MeasureSpec.AT_MOST), MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED))
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
        setMeasuredDimension(MeasureSpec.getSize(wSpec), y + lineH + paddingTop + paddingBottom)
    }

    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val maxW = r - l - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            c.layout(paddingLeft + x, paddingTop + y, paddingLeft + x + c.measuredWidth, paddingTop + y + c.measuredHeight)
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
    }
}

package com.azkar.app.ui

import android.text.Editable
import android.text.TextWatcher
import android.view.View
import android.view.inputmethod.EditorInfo
import android.widget.EditText
import com.azkar.app.R
import com.azkar.app.data.Store

/**
 * Shared behaviour of the Add zikr and Edit zikr screens:
 * repeat stepper and folder chips.
 */
class ZikrForm(
    private val view: View,
    private val allowNew: Boolean,
    private val onSelChanged: (String?) -> Unit = {}
) {
    /** Last valid repeat count. */
    private var count = 1

    /** Prevents TextWatcher recursion during programmatic updates. */
    private var syncing = false

    var n: Int
        get() = box.text.toString()
            .toIntOrNull()
            ?.takeIf { it >= 1 }
            ?.coerceAtMost(999) ?: count
        set(v) {
            count = v.coerceIn(1, 999)
            syncBox()
        }

    var sel: String? = null
        set(v) {
            field = v
            renderChips()
            onSelChanged(v)
        }

    val arabic: EditText = view.findViewById(R.id.ar)
    private val box: EditText = view.findViewById(R.id.n)

    init {
        view.findViewById<View>(R.id.btn_dec).setOnClickListener {
            n = n - 1
        }

        view.findViewById<View>(R.id.btn_inc).setOnClickListener {
            n = n + 1
        }

        box.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(
                s: CharSequence?, start: Int, count: Int, after: Int
            ) {}

            override fun onTextChanged(
                s: CharSequence?, start: Int, before: Int, count: Int
            ) {}

            override fun afterTextChanged(s: Editable?) {
                if (syncing) return

                s?.toString()
                    ?.toIntOrNull()
                    ?.takeIf { it >= 1 }
                    ?.let { count = it.coerceAtMost(999) }

                updateDec()
            }
        })

        box.setOnFocusChangeListener { _, focused ->
            if (!focused) syncBox()
        }

        box.setOnEditorActionListener { _, action, _ ->
            if (action == EditorInfo.IME_ACTION_DONE) {
                syncBox()
            }
            false
        }

        syncBox()
    }

    private fun syncBox() {
        syncing = true
        try {
            val value = count.toString()
            if (box.text.toString() != value) {
                box.setText(value)
            }
        } finally {
            syncing = false
        }

        updateDec()
    }

    private fun updateDec() {
        view.findViewById<View>(R.id.btn_dec).isEnabled = n > 1
    }

    fun renderChips() {
        val ctx = view.context
        val chips = view.findViewById<FlowLayout>(R.id.chips)

        chips.removeAllViews()

        val choices = Store.folders().map {
            it.id to ((it.icon?.let { e -> "$e " } ?: "") + it.name)
        }

        choices.forEach { (k, label) ->
            chips.addView(
                Ui.chip(ctx, label, sel == k, false) {
                    sel = k
                }
            )
        }

        if (allowNew) {
            chips.addView(
                Ui.chip(ctx, "+ New folder", sel == "new", true) {
                    sel = "new"
                }
            )
        }

const val ACTION_COUNT = "com.azkar.app.widget.COUNT"
const val ACTION_RESET = "com.azkar.app.widget.RESET"
const val ACTION_REFRESH = "com.azkar.app.widget.REFRESH"

abstract class BaseWidgetProvider(private val kind: WidgetKind) : AppWidgetProvider() {

    override fun onUpdate(ctx: Context, mgr: AppWidgetManager, ids: IntArray) {
        Store.init(ctx)
        val pending = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Store.awaitReady()
                ids.forEach { WidgetUpdater.update(ctx, mgr, it, kind) }
            } finally {
                pending.finish()
            }
        }
    }

    override fun onAppWidgetOptionsChanged(ctx: Context, mgr: AppWidgetManager, id: Int, newOptions: Bundle) {
        Store.init(ctx)
        val pending = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Store.awaitReady()
                WidgetUpdater.update(ctx, mgr, id, kind)
            } finally {
                pending.finish()
            }
        }
    }

    override fun onDeleted(ctx: Context, ids: IntArray) {
        ids.forEach { WidgetStore.clear(ctx, it) }
    }

    override fun onReceive(ctx: Context, intent: Intent) {
        if (intent.action == Intent.ACTION_DATE_CHANGED ||
            intent.action == Intent.ACTION_TIME_CHANGED ||
            intent.action == Intent.ACTION_TIMEZONE_CHANGED ||
            intent.action == Intent.ACTION_BOOT_COMPLETED
        ) {
            Store.init(ctx)
            dispatch(ctx) { WidgetUpdater.updateAll(ctx) }
            return
        }
        val id = intent.getIntExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, AppWidgetManager.INVALID_APPWIDGET_ID)
        when (intent.action) {
            ACTION_COUNT, ACTION_RESET -> {
                if (id == AppWidgetManager.INVALID_APPWIDGET_ID) return
                Store.init(ctx)
                dispatch(ctx) {
                    if (intent.action == ACTION_COUNT) WidgetLogic.tap(ctx, id, kind) else WidgetLogic.reset(ctx, id)
                    WidgetUpdater.update(ctx, AppWidgetManager.getInstance(ctx), id, kind)
                }
            }
            ACTION_REFRESH -> {
                Store.init(ctx)
                dispatch(ctx) { WidgetUpdater.updateAll(ctx) }
            }
            else -> super.onReceive(ctx, intent)
        }
    }

    private fun dispatch(ctx: Context, block: suspend () -> Unit) {
        val pending = goAsync()
        CoroutineScope(Dispatchers.IO).launch {
            try {
                Store.awaitReady()
                block()
            } finally {
                pending.finish()
            }
        }
    }
}

object WidgetUpdater {
    private val DEFAULT_DP = mapOf(
        WidgetKind.SMALL to (158f to 158f),
        WidgetKind.MEDIUM to (338f to 158f),
        WidgetKind.LARGE to (338f to 354f),
    )

    fun providerClass(kind: WidgetKind): Class<*> = when (kind) {
        WidgetKind.SMALL -> SmallWidgetProvider::class.java
        WidgetKind.MEDIUM -> MediumWidgetProvider::class.java
        WidgetKind.LARGE -> LargeWidgetProvider::class.java
    }

    fun kindOf(ctx: Context, id: Int): WidgetKind {
        val cn = AppWidgetManager.getInstance(ctx).getAppWidgetInfo(id)?.provider?.className
        return WidgetKind.values().firstOrNull { providerClass(it).name == cn } ?: WidgetKind.SMALL
    }

    /** Re-draws every widget (call it when the app's data may have changed, e.g. MainActivity.onStop). */
    fun updateAll(ctx: Context) {
        val mgr = AppWidgetManager.getInstance(ctx)
        WidgetKind.values().forEach { k ->
            mgr.getAppWidgetIds(ComponentName(ctx, providerClass(k))).forEach { update(ctx, mgr, it, k) }
        }
    }

    fun update(ctx: Context, mgr: AppWidgetManager, id: Int, kind: WidgetKind) {
        val cfg = WidgetStore.config(ctx, id, kind)
        WidgetStore.scheduleRefresh(ctx, id, cfg.refresh)
        val model = WidgetModel.load(ctx, id, cfg.folder)

        // Real size of this widget on the launcher grid, in dp.
        val o = mgr.getAppWidgetOptions(id)
        val land = ctx.resources.configuration.orientation == Configuration.ORIENTATION_LANDSCAPE
        val def = DEFAULT_DP.getValue(kind)
        val w = (if (land) o.getInt(AppWidgetManager.OPTION_APPWIDGET_MAX_WIDTH) else o.getInt(AppWidgetManager.OPTION_APPWIDGET_MIN_WIDTH))
            .takeIf { it > 0 }?.toFloat() ?: def.first
        val h = (if (land) o.getInt(AppWidgetManager.OPTION_APPWIDGET_MIN_HEIGHT) else o.getInt(AppWidgetManager.OPTION_APPWIDGET_MAX_HEIGHT))
            .takeIf { it > 0 }?.toFloat() ?: def.second

        val bmp = WidgetRenderer(ctx, cfg.theme).render(cfg.style, kind, model, w, h, ctx.resources.displayMetrics.density)

        val layout = when {
            cfg.style == WidgetStyle.CLASSIC && kind == WidgetKind.LARGE -> R.layout.widget_classic_large
            cfg.style == WidgetStyle.TASBIH && kind == WidgetKind.LARGE -> R.layout.widget_tasbih_large
            else -> R.layout.widget_plain
        }
        val rv = RemoteViews(ctx.packageName, layout)
        rv.setImageViewBitmap(R.id.img, bmp)
        rv.setContentDescription(
            R.id.img,
            if (model.n == 0) ctx.getString(R.string.widget_no_azkar_description)
            else "${model.folderName}: ${model.shown} of ${model.target}"
        )
        rv.setContentDescription(R.id.hit_open, ctx.getString(R.string.widget_action_open))
        rv.setContentDescription(R.id.hit_count, ctx.getString(R.string.widget_action_count))

        val primary = if (cfg.tapAction == WidgetStore.TapAction.COUNT) {
            action(ctx, id, kind, ACTION_COUNT)
        } else {
            openApp(ctx, id, cfg.folder)
        }
        rv.setOnClickPendingIntent(R.id.hit_open, primary)
        val count = action(ctx, id, kind, ACTION_COUNT)
        rv.setOnClickPendingIntent(R.id.hit_count, count)
        if (layout == R.layout.widget_classic_large) {
            rv.setContentDescription(R.id.hit_count2, ctx.getString(R.string.widget_action_count))
            rv.setContentDescription(R.id.btn_plus, ctx.getString(R.string.widget_action_count))
            rv.setOnClickPendingIntent(R.id.hit_count2, count)
            rv.setOnClickPendingIntent(R.id.btn_plus, count)
        }

        if (layout == R.layout.widget_tasbih_large) {
            rv.setContentDescription(R.id.btn_main, ctx.getString(R.string.widget_action_count))
            rv.setContentDescription(R.id.btn_reset, ctx.getString(R.string.widget_action_reset))
            rv.setOnClickPendingIntent(R.id.btn_main, count)
            rv.setOnClickPendingIntent(R.id.btn_reset, action(ctx, id, kind, ACTION_RESET))
        }

        mgr.updateAppWidget(id, rv)
    }

    private fun action(ctx: Context, id: Int, kind: WidgetKind, action: String): PendingIntent {
        val i = Intent(ctx, providerClass(kind)).setAction(action)
            .setData(Uri.parse("azkar://widget/$id/$action"))
            .putExtra(AppWidgetManager.EXTRA_APPWIDGET_ID, id)
        return PendingIntent.getBroadcast(ctx, id * 2 + (if (action == ACTION_COUNT) 0 else 1), i,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)
    }

    /** Opens the app straight into this folder's reader (MainActivity already handles the "open" extra). */
    private fun openApp(ctx: Context, id: Int, folder: String): PendingIntent {
        val i = Intent(ctx, MainActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
            .putExtra("open", folder)
        return PendingIntent.getActivity(ctx, 100000 + id, i, PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)
    }

package com.azkar.app.widgets

import android.content.Context

enum class WidgetKind { SMALL, MEDIUM, LARGE }

private val ALL_KINDS = setOf(WidgetKind.SMALL, WidgetKind.MEDIUM, WidgetKind.LARGE)

/** The four available designs. [kinds] = which sizes exist for that design. */
enum class WidgetStyle(val label: String, val kinds: Set<WidgetKind>) {
    CLASSIC("Classic", ALL_KINDS),
    CHECKLIST("Checklist", setOf(WidgetKind.MEDIUM, WidgetKind.LARGE)),
    TASBIH("Tasbih beads", setOf(WidgetKind.SMALL, WidgetKind.LARGE)),
    CALLIGRAPHY("Calligraphy", setOf(WidgetKind.SMALL, WidgetKind.MEDIUM));
}

/** Everything a renderer needs for one widget, already read from [Store]. */
class WidgetModel(
    val folderName: String,
    val items: List<ReaderItem>,
    val labels: List<String>,   // short left-hand label per zikr (its note, or "×N")
    val idx: Int,               // index of the zikr being recited; == n when the whole folder is finished
    val count: Int,             // recitations done on the current zikr
    val date: String,
) {
    val n = items.size
    val finished = n > 0 && idx >= n
    val curIdx = if (n == 0) 0 else min(idx, n - 1)
    val cur: ReaderItem? = items.getOrNull(curIdx)
    val target = cur?.target ?: 1
    val shown = if (finished) target else count.coerceIn(0, target)
    val done = if (finished) n else idx
    val position = if (n == 0) 0 else min(idx + 1, n)

    // Tasbih beads: one ring = up to 33 beads
    val beadTotal = min(33, target).coerceAtLeast(1)
    val beadFilled = when {
        shown == 0 -> 0
        shown % beadTotal == 0 -> beadTotal
        else -> shown % beadTotal
    }
    val round = if (shown == 0) 1 else (shown - 1) / beadTotal + 1
    val rounds = ceil(target / beadTotal.toDouble()).toInt().coerceAtLeast(1)

    companion object {
        fun load(ctx: Context, widgetId: Int, folder: String): WidgetModel {
            val items = Store.readerItems(folder)
            val (i, c) = WidgetStore.progress(ctx, widgetId)
            val idx = i.coerceIn(0, items.size)
            val cnt = if (idx < items.size) c.coerceIn(0, (items[idx].target - 1).coerceAtLeast(0)) else 0
            val labels = items.map { it0 ->
                Store.zikr(it0.id)?.note?.trim()?.takeIf { s -> s.isNotEmpty() } ?: "×${it0.target}"
            }
            return WidgetModel(
                Store.folderName(folder), items, labels, idx, cnt,
                DateFormat.format("EEE, d MMM", Date()).toString()
            )
        }
    }
}

 */
class WidgetRenderer(ctx: Context, theme: Int? = null) {

    }

    private val palette = WidgetPalette.from(ctx, (theme ?: Store.theme()) == com.azkar.app.ThemeMode.OLED)

    private val CARD_BORDER get() = palette.border

    private val GOLD_SOFT get() = palette.accentSoft

    fun render(style: WidgetStyle, kind: WidgetKind, m: WidgetModel, wDp: Float, hDp: Float, density: Float): Bitmap {
        val widthDp = wDp.coerceAtLeast(1f)
        val heightDp = hDp.coerceAtLeast(1f)
        var s = density
        val px = widthDp * heightDp * s * s
        if (px > MAX_PIXELS) s *= sqrt(MAX_PIXELS / px)       // RemoteViews bitmaps must stay small
        val bmp = Bitmap.createBitmap(max(1, (widthDp * s).roundToInt()), max(1, (heightDp * s).roundToInt()), Bitmap.Config.ARGB_8888)
        val c = Canvas(bmp)
        c.scale(s, s)
        if (m.n == 0) { empty(c, widthDp, heightDp); return bmp }
        when (style) {
            WidgetStyle.CLASSIC -> when (kind) {
                WidgetKind.SMALL -> classicSmall(c, widthDp, heightDp, m)
                WidgetKind.MEDIUM -> classicMedium(c, widthDp, heightDp, m)
                WidgetKind.LARGE -> classicLarge(c, widthDp, heightDp, m)

            WidgetStyle.CALLIGRAPHY -> if (kind == WidgetKind.SMALL) calligraphySmall(c, widthDp, heightDp, m) else calligraphyMedium(c, widthDp, heightDp, m)
        }
        return bmp
    }

    private fun strokeP(color: Int, w: Float, cap: Paint.Cap = Paint.Cap.BUTT): Paint {
        p.reset(); p.isAntiAlias = true; p.style = Paint.Style.STROKE; p.color = color; p.strokeWidth = w; p.strokeCap = cap; return p

    /** Latin UI text, vertically centred on [cy]. */
    private fun text(
        c: Canvas, s: String, x: Float, cy: Float, size: Float, color: Int, tf: Typeface,
        align: Paint.Align = Paint.Align.LEFT, spacing: Float = 0f, maxW: Float = 0f,
    ) {
        tp.reset(); tp.isAntiAlias = true
        tp.textSize = size; tp.color = color; tp.typeface = tf; tp.textAlign = align; tp.letterSpacing = spacing
        var str = s
        if (maxW > 0f && tp.measureText(str) > maxW) str = TextUtils.ellipsize(s, tp, maxW, TextUtils.TruncateAt.END).toString()
        val fm = tp.fontMetrics
        c.drawText(str, x, cy - (fm.ascent + fm.descent) / 2f, tp)
    }

    private fun layout(s: String, w: Int, size: Float, lh: Float, color: Int, bold: Boolean, align: Layout.Alignment, maxLines: Int): StaticLayout {
        val t = arabicPaint
        t.reset()
        t.typeface = if (bold) amiriBold else amiriReg
        t.textSize = size; t.color = color
        val fm = t.fontMetrics
        val b = StaticLayout.Builder.obtain(s, 0, s.length, t, w)
            .setAlignment(align)
            .setTextDirection(TextDirectionHeuristics.RTL)
            .setIncludePad(false)
            .setLineSpacing(size * lh - (fm.descent - fm.ascent), 1f)
        if (maxLines > 0) b.setMaxLines(maxLines).setEllipsize(TextUtils.TruncateAt.END).setEllipsizedWidth(w)
        return b.build()
    }

    /** Arabic paragraph in a box. Shrinks the font until it fits. */
    private fun arabic(
        c: Canvas, s: String, l: Float, t: Float, r: Float, b: Float, size: Float, lh: Float, color: Int,
        bold: Boolean = true, align: Layout.Alignment = Layout.Alignment.ALIGN_CENTER, v: V = V.CENTER,
    ) {
        val w = max(1, (r - l).toInt())
        var sz = size
        var lay = layout(s, w, sz, lh, color, bold, align, 0)
        while (lay.height > (b - t) && sz > size * 0.5f) { sz *= 0.93f; lay = layout(s, w, sz, lh, color, bold, align, 0) }
        val y = when (v) { V.TOP -> t; V.CENTER -> t + ((b - t) - lay.height) / 2f; V.BOTTOM -> b - lay.height }
        c.save(); c.translate(l, y); lay.draw(c); c.restore()
    }

    /** One Arabic line, right-aligned, ellipsized, centred on [cy]. */
    private fun arabicLine(c: Canvas, s: String, l: Float, r: Float, cy: Float, size: Float, color: Int) {
        val lay = layout(s, max(1, (r - l).toInt()), size, 1.2f, color, false, Layout.Alignment.ALIGN_NORMAL, 1)
        c.save(); c.translate(l, cy - lay.height / 2f); lay.draw(c); c.restore()
    }

package com.azkar.app.widgets

import android.content.Context
import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Intent
import android.net.Uri
import com.azkar.app.data.Store
import com.azkar.app.R
import android.util.Log
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/** Per-widget settings and progress. Progress resets itself every new day. */
object WidgetStore {
    private fun sp(ctx: Context) = ctx.getSharedPreferences("azkar_widgets", Context.MODE_PRIVATE)
    private fun today() = SimpleDateFormat("yyyyMMdd", Locale.US).format(Date())

    enum class Refresh(val minutes: Long, val label: Int) {
        ON_CHANGE(0, R.string.widget_refresh_on_change),
        HOURLY(60, R.string.widget_refresh_hourly),
        EVERY_SIX_HOURS(360, R.string.widget_refresh_six_hours),
        DAILY(1440, R.string.widget_refresh_daily)
    }

    enum class TapAction(val label: Int) {
        OPEN(R.string.widget_tap_open), COUNT(R.string.widget_tap_count)
    }

    data class Config(
        val style: WidgetStyle,
        val folder: String,
        val theme: Int?,
        val refresh: Refresh,
        val tapAction: TapAction
    )

    fun config(ctx: Context, id: Int, kind: WidgetKind): Config {
        val s = sp(ctx)
        val savedStyle = s.getString("style_$id", null)
        val style = try {
            savedStyle?.let(WidgetStyle::valueOf)
        } catch (e: IllegalArgumentException) {
            Log.w("WidgetStore", "Invalid widget style '$savedStyle' for $id", e)
            null
        }
            ?.takeIf { kind in it.kinds } ?: WidgetStyle.CLASSIC
        val folder = s.getString("folder_$id", null) ?: Store.folders().firstOrNull()?.id ?: "anytime"
        val theme = if (s.contains("theme_$id")) s.getInt("theme_$id", com.azkar.app.ThemeMode.CURRENT) else null
        val refresh = runCatching { Refresh.valueOf(s.getString("refresh_$id", Refresh.ON_CHANGE.name)!!) }
            .getOrDefault(Refresh.ON_CHANGE)
        val tap = runCatching { TapAction.valueOf(s.getString("tap_$id", TapAction.OPEN.name)!!) }
            .getOrDefault(TapAction.OPEN)
        return Config(style, folder, theme, refresh, tap)
    }

    fun setConfig(
        ctx: Context,
        id: Int,
        style: WidgetStyle,
        folder: String,
        theme: Int?,
        refresh: Refresh = Refresh.ON_CHANGE,
        tapAction: TapAction = TapAction.OPEN
    ) {
        val edit = sp(ctx).edit()
            .putString("style_$id", style.name)
            .putString("folder_$id", folder)
            .putString("refresh_$id", refresh.name)
            .putString("tap_$id", tapAction.name)
        if (theme == null) edit.remove("theme_$id") else edit.putInt("theme_$id", theme)
        edit.commit()
        saveProgress(ctx, id, 0, 0)
        scheduleRefresh(ctx, id, refresh)
    }

    /** (index, count) for today; (0, 0) if the saved progress is from a previous day. */
    fun progress(ctx: Context, id: Int): Pair<Int, Int> {
        val s = sp(ctx)
        if (s.getString("day_$id", "") != today()) return 0 to 0
        return s.getInt("idx_$id", 0) to s.getInt("cnt_$id", 0)
    }

    fun saveProgress(ctx: Context, id: Int, idx: Int, count: Int) {
        sp(ctx).edit().putString("day_$id", today()).putInt("idx_$id", idx).putInt("cnt_$id", count).apply()
    }

    fun clear(ctx: Context, id: Int) {
        cancelRefresh(ctx, id)
        sp(ctx).edit().remove("style_$id").remove("folder_$id").remove("theme_$id").remove("day_$id").remove("idx_$id").remove("cnt_$id").apply()
    }

    fun scheduleRefresh(ctx: Context, id: Int, refresh: Refresh) {
        cancelRefresh(ctx, id)
        if (refresh == Refresh.ON_CHANGE) return
        val alarm = ctx.getSystemService(AlarmManager::class.java)
        val pi = refreshPending(ctx, id, true)!!
        alarm.setInexactRepeating(
            AlarmManager.RTC,
            System.currentTimeMillis() + refresh.minutes * 60_000L,
            refresh.minutes * 60_000L,
            pi
        )
    }

    fun cancelRefresh(ctx: Context, id: Int) {
        val alarm = ctx.getSystemService(AlarmManager::class.java)
        refreshPending(ctx, id)?.let { alarm.cancel(it); it.cancel() }
    }

    private fun refreshPending(ctx: Context, id: Int, create: Boolean = false): PendingIntent? =
        PendingIntent.getBroadcast(
            ctx, id, refreshIntent(ctx, id),
            (if (create) PendingIntent.FLAG_UPDATE_CURRENT else PendingIntent.FLAG_NO_CREATE) or PendingIntent.FLAG_IMMUTABLE
        )

    private fun refreshIntent(ctx: Context, id: Int): Intent =
        Intent(ctx, WidgetRefreshReceiver::class.java).setData(Uri.parse("azkar://widget-refresh/$id"))
            .putExtra(android.appwidget.AppWidgetManager.EXTRA_APPWIDGET_ID, id)
}

/** What a tap does. */
object WidgetLogic {
    /** +1 on the current zikr; moves on when it is complete; after the last one the next tap starts over. */
    fun tap(ctx: Context, id: Int, kind: WidgetKind) {
        val cfg = WidgetStore.config(ctx, id, kind)
        val items = Store.readerItems(cfg.folder)
        if (items.isEmpty()) return
        var (idx, cnt) = WidgetStore.progress(ctx, id)
        idx = idx.coerceIn(0, items.size)
        if (idx >= items.size) { idx = 0; cnt = 0 }
        else {
            cnt++
            if (cnt >= items[idx].target) { idx++; cnt = 0 }
        }
        WidgetStore.saveProgress(ctx, id, idx, cnt)
    }

    fun reset(ctx: Context, id: Int) = WidgetStore.saveProgress(ctx, id, 0, 0)
}

package com.azkar.app.reminders

import org.junit.Assert.assertTrue
import org.junit.Test
import java.util.Calendar

class ReminderTimeTest {
    @Test
    fun schedulesTomorrowWhenTodaysTimeHasPassed() {
        val now = Calendar.getInstance().apply {
            set(2026, Calendar.JANUARY, 10, 18, 0, 0)
            set(Calendar.MILLISECOND, 0)
        }
        val next = Calendar.getInstance().apply { timeInMillis = ReminderTime.next(17, 0, now.timeInMillis) }
        assertTrue(next.after(now))
        assertTrue(next.get(Calendar.DAY_OF_YEAR) != now.get(Calendar.DAY_OF_YEAR))
    }

    @Test
    fun preservesValidLocalTimeAcrossDaylightSavingTransitions() {
        val now = Calendar.getInstance().apply {
            set(2026, Calendar.MARCH, 28, 23, 0, 0)
            set(Calendar.MILLISECOND, 0)
        }
        val next = Calendar.getInstance().apply { timeInMillis = ReminderTime.next(8, 0, now.timeInMillis) }
        assertTrue(next.get(Calendar.HOUR_OF_DAY) == 8)
        assertTrue(next.after(now))
    }
}

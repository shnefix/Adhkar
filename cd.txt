        var i = 0

package com.azkar.app.data

import org.json.JSONArray
import org.json.JSONObject

/** Builds a backup file that ImportJson reads back. Read-only: never touches stored data. */
object ExportJson {
    private fun item(z: Zikr) = JSONObject().put("text", z.arabic).put("count", z.repeat).also { if (z.note.isNotBlank()) it.put("note", z.note) }

    fun build(s: ImportMerge.State, arabicSize: Int, notificationsOn: Boolean, arabicSp: Int = ReaderPrefs.FONT_DEFAULT): String {
        val folders = JSONArray()
        s.folders.forEach { f ->
            val az = JSONArray()
            s.zikrs.filter { it.folderId == f.id }.forEach { az.put(item(it)) }
            val o = JSONObject().put("id", f.id).put("name", f.name)
            f.icon?.let { o.put("icon", it) }
if (f.remember) o.put("remember", true)
            folders.put(o.put("azkar", az))
        }
        val root = JSONObject().put("app", "azkar").put("version", 1).put("folders", folders)
        val unfiled = JSONArray()
        s.zikrs.filter { it.folderId == null }.forEach { unfiled.put(item(it)) }
        if (unfiled.length() > 0) root.put("unfiled", unfiled)
        val rems = JSONArray()
        s.schedules.forEach {
            val o = JSONObject().put("id", it.id).put("title", it.title).put("message", it.message)
                .put("time", "%02d:%02d".format(it.hour, it.minute)).put("enabled", it.enabled)
            it.open?.let { f -> o.put("folder", f) }
            rems.put(o)
        }
        root.put("reminders", rems)
        root.put("settings", JSONObject().put("arabicSize", arabicSize).put("notificationsEnabled", notificationsOn).put("arabicSp", arabicSp))
        return root.toString(2)
    }
}

===== app\src\main\java\com\azkar\app\data\ImportJson.kt =====
package com.azkar.app.data

import org.json.JSONException
import org.json.JSONObject

/**
 * Parses and validates an Azkar JSON file (hand-written or produced by Export).
 * Only "folders" is required. Optional: folder "id"/"icon", zikr "note", top-level "unfiled", "reminders", "settings".
 */
object ImportJson {
    data class Item(val text: String, val count: Int, val note: String = "")
    data class FolderIn(val name: String, val items: List<Item>, val id: String? = null, val icon: String? = null, val remember: Boolean = false, val home: Boolean? = null)
    data class ReminderIn(val id: String, val title: String, val message: String, val hour: Int, val minute: Int, val enabled: Boolean, val folder: String?)
    data class SettingsIn(val arabicSize: Int?, val notificationsOn: Boolean?, val arabicSp: Int? = null)
    sealed class Parsed {
        data class Ok(
            val folders: List<FolderIn>,
            val unfiled: List<Item> = emptyList(),
            val reminders: List<ReminderIn> = emptyList(),
            val settings: SettingsIn? = null
        ) : Parsed()
        data class Error(val message: String) : Parsed()
    }

    private val TIME = Regex("^(\\d{1,2}):(\\d{2})$")

    private fun items(az: org.json.JSONArray, where: String): Any {
        val out = ArrayList<Item>()
        for (j in 0 until az.length()) {
            val o = az.optJSONObject(j) ?: return "$where, zikr ${j + 1}: expected an object."
            val text = if (o.isNull("text")) "" else o.optString("text", "")
            if (text.isBlank()) return "$where, zikr ${j + 1}: missing \"text\"."
            val n = when (val c = o.opt("count")) {
                null, JSONObject.NULL -> 1
                is Int -> c
                is Long -> if (c in 1..999) c.toInt() else -1
                is Double -> if (c % 1.0 == 0.0) c.toInt() else -1
                is String -> c.trim().toIntOrNull() ?: -1
                else -> -1
            }
            if (n < 1 || n > 999) return "$where, zikr ${j + 1}: \"count\" must be a whole number from 1 to 999."
            val note = if (o.isNull("note")) "" else o.optString("note", "")
            out.add(Item(text, n, note))
        }
        return out
    }

    @Suppress("UNCHECKED_CAST")
    fun parse(raw: String): Parsed {
        val s = raw.removePrefix("\uFEFF").trim()
        if (s.isEmpty()) return Parsed.Error("The file is empty.")
        if (s.startsWith("[")) return Parsed.Error("The file is a bare list. It must be an object like { \"folders\": [ ... ] }.")
        val root = try { JSONObject(s) } catch (e: JSONException) {
            return Parsed.Error("This is not valid JSON. " + (e.message ?: "").take(90))
        }
        val arr = root.optJSONArray("folders") ?: return Parsed.Error(
            if (root.has("folders")) "\"folders\" must be a list, like [ { \"name\": ..., \"azkar\": [...] } ]."
            else "The file needs a top-level \"folders\" list."
        )
        val folders = ArrayList<FolderIn>()
        for (i in 0 until arr.length()) {
            val f = arr.optJSONObject(i) ?: return Parsed.Error("Folder ${i + 1}: expected an object.")
            val name = if (f.isNull("name")) "" else f.optString("name", "").trim()
            if (name.isEmpty()) return Parsed.Error("Folder ${i + 1}: missing \"name\".")
            val az = f.optJSONArray("azkar") ?: return Parsed.Error("Folder \"$name\": missing an \"azkar\" list.")
            val r = items(az, "Folder \"$name\"")
            if (r is String) return Parsed.Error(r)
            val id = if (f.isNull("id")) null else (f.opt("id") as? String)?.trim()?.ifEmpty { null }
            val icon = (if (f.isNull("icon")) null else (f.opt("icon") as? String)?.trim()?.ifEmpty { null })
    ?.takeUnless { EmojiRules.isRestricted(it) }   // people/animal emojis are dropped
            folders.add(FolderIn(name, r as List<Item>, id, icon, f.optBoolean("remember", false), if (f.opt("home") is Boolean) f.getBoolean("home") else null))
        }
        var unfiled: List<Item> = emptyList()
        root.optJSONArray("unfiled")?.let {
            val r = items(it, "Unfiled azkar")
            if (r is String) return Parsed.Error(r)
            unfiled = r as List<Item>
        }
        val rems = ArrayList<ReminderIn>()
        root.optJSONArray("reminders")?.let { ra ->
            for (i in 0 until ra.length()) {
                val o = ra.optJSONObject(i) ?: return Parsed.Error("Reminder ${i + 1}: expected an object.")
                val title = if (o.isNull("title")) "" else o.optString("title", "").trim()
                if (title.isEmpty()) return Parsed.Error("Reminder ${i + 1}: missing \"title\".")
                val m = TIME.matchEntire(if (o.isNull("time")) "" else o.optString("time", "").trim())
                val h = m?.groupValues?.get(1)?.toInt() ?: -1
                val mi = m?.groupValues?.get(2)?.toInt() ?: -1
                if (h !in 0..23 || mi !in 0..59) return Parsed.Error("Reminder \"$title\": \"time\" must look like 06:30.")
                val msg = if (o.isNull("message")) "" else o.optString("message", "")
                val id = if (o.isNull("id")) "" else o.optString("id", "").trim()
                val folder = if (o.isNull("folder")) null else (o.opt("folder") as? String)?.trim()?.ifEmpty { null }
                rems.add(ReminderIn(id, title, msg.ifBlank { "Time for your azkar · tap to read" }, h, mi, o.optBoolean("enabled", true), folder))
            }
        }
        val st = root.optJSONObject("settings")?.let { so ->
            SettingsIn(
                (so.opt("arabicSize") as? Number)?.toInt()?.takeIf { it in 0..2 },
                if (so.opt("notificationsEnabled") is Boolean) so.getBoolean("notificationsEnabled") else null,
                (so.opt("arabicSp") as? Number)?.toInt()?.takeIf { it in ReaderPrefs.FONT_MIN..ReaderPrefs.FONT_MAX }
            )
        }
        if (folders.sumOf { it.items.size } + unfiled.size == 0 && rems.isEmpty()) {
            return Parsed.Error(if (folders.isEmpty()) "The \"folders\" list is empty." else "The file contains no azkar.")
        }
        return Parsed.Ok(folders, unfiled, rems, st)
    }
}

===== app\src\main\java\com\azkar\app\data\ImportMerge.kt =====
package com.azkar.app.data

/** Pure merge of an import into the current data. Never overwrites or deletes anything that exists. */
object ImportMerge {
    data class State(val folders: List<Folder>, val zikrs: List<Zikr>, val schedules: List<Schedule>)
    data class Summary(val created: Int, val merged: Int, val added: Int, val skipped: Int, val reminders: Int)

    fun apply(s: State, p: ImportJson.Parsed.Ok, newId: () -> String): Pair<State, Summary> {
        val fl = s.folders.toMutableList()
        val zl = s.zikrs.toMutableList()
        val sl = s.schedules.toMutableList()
        val idMap = HashMap<String, String>()          // folder id in the file -> folder id in the app
        var created = 0; var merged = 0; var added = 0; var skipped = 0; var rem = 0

        fun addItems(folderId: String?, items: List<ImportJson.Item>) {
            items.forEach { item ->
                if (zl.any { it.folderId == folderId && it.arabic == item.text }) skipped++
                else { zl.add(Zikr(newId(), item.text, item.note, item.count, folderId)); added++ }
            }
        }

        p.folders.forEach { f ->
            val byId = f.id?.let { id -> fl.firstOrNull { it.id == id } }
            val existing = byId ?: fl.firstOrNull { it.name.trim().equals(f.name.trim(), ignoreCase = true) }
            val folder = existing ?: run {
                val id = f.id?.takeIf { c -> c != "anytime" && !c.startsWith("zikr:") && fl.none { it.id == c } } ?: newId()
                Folder(id, f.name.trim(), f.icon, f.remember).also { fl.add(it); created++ }
            }
            if (existing != null) merged++
            f.id?.let { idMap[it] = folder.id }
            addItems(folder.id, f.items)
        }
        addItems(null, p.unfiled)
        p.reminders.forEach { r ->
            val id = r.id.ifBlank { newId() }
            if (sl.none { it.id == id }) {
                val target = r.folder?.let { ref -> idMap[ref] ?: fl.firstOrNull { it.id == ref }?.id }
                sl.add(Schedule(id, r.title, r.message, r.hour, r.minute, r.enabled, target)); rem++
            }
        }
        return State(fl, zl, sl) to Summary(created, merged, added, skipped, rem)
    }
}

package com.azkar.app.data

/** Reader constants and pure rules that do not need Android. */
object ReaderPrefs {
    const val FONT_MIN = 18
    const val FONT_MAX = 40
    const val FONT_DEFAULT = 26            // the old "Medium"
    private val OLD_SIZES = intArrayOf(20, 26, 32)   // old Small / Medium / Large

    fun spFromOldIndex(i: Int) = OLD_SIZES[i.coerceIn(0, 2)]
    fun clampSp(sp: Int) = sp.coerceIn(FONT_MIN, FONT_MAX)
}

/** Where a folder should start reading when it is opened. */
object ReaderResume {
    /** (index, count). Falls back to the beginning if the saved zikr is gone, hidden or the id is unknown. */
    fun start(ids: List<String>, targets: List<Int>, savedId: String?, savedCount: Int): Pair<Int, Int> {
        if (savedId == null) return 0 to 0
        val i = ids.indexOf(savedId)
        if (i < 0 || i >= targets.size) return 0 to 0
        return i to savedCount.coerceIn(0, targets[i])
    }
}

    private const val K_SP = "arabic_sp"
    private const val K_TAP = "tap_advance"
    private const val K_AWAKE = "keep_awake"
    private const val K_POS = "pos_"      // pos_<folderId> = last zikr id
    private const val K_POSC = "posc_"    // posc_<folderId> = its count

    // In-memory copies: the single source of truth for every screen.
    private var zCache: List<Zikr> = emptyList()
    private var fCache: List<Folder> = emptyList()
    private var sCache: List<Schedule> = emptyList()
    private var hzCache: Set<String> = emptySet()

    @Synchronized
    fun init(context: Context) {
        if (::sp.isInitialized) return
        appCtx = context.applicationContext
        sp = appCtx.getSharedPreferences("azkar", Context.MODE_PRIVATE)
        zCache = parseZikrs(sp.getString(K_ZIKRS, "[]") ?: "[]")
        fCache = parseFolders(sp.getString(K_FOLDERS, "[]") ?: "[]")
        sCache = parseSchedules(sp.getString(K_SCHED, "[]") ?: "[]")
        hzCache = parseIds(sp.getString(K_HZ, "[]") ?: "[]")
        syncDefaults()
        migrateSchedules()
        sanitizeIcons()
    }

    // ---------- Default azkar (assets/default_folders.json) ----------
    private fun defaults(): ImportJson.Parsed.Ok? = try {
        val raw = appCtx.assets.open("default_folders.json").bufferedReader(Charsets.UTF_8).use { it.readText() }
        ImportJson.parse(raw) as? ImportJson.Parsed.Ok
    } catch (e: Exception) {
        android.util.Log.w("Azkar", "default folders were not loaded", e)
        null
    }

    /**
     * Gives every user the defaults from the JSON, additively: folders match by id then name,
     * azkar are skipped if the same text already exists in that folder, nothing is overwritten or deleted.
     * Runs once per DEFAULTS_VERSION (bump it when you add defaults). Users of the previous build already
     * received everything, so for them it only records the version.
     */
    private fun syncDefaults() {
        if (sp.getInt(K_DEFAULTS_VER, 0) >= DEFAULTS_VERSION) return
        val alreadyHave = sp.getBoolean(K_DEFAULT_FOLDERS_IMPORTED, false)
        if (!alreadyHave && !mergeDefaults()) return          // asset unreadable: try again next start
        sp.edit().putInt(K_DEFAULTS_VER, DEFAULTS_VERSION).apply()
    }

    private fun mergeDefaults(): Boolean {
        val p = defaults() ?: return false
        val before = fCache.map { it.id }.toSet()
        val (ns, _) = ImportMerge.apply(ImportMerge.State(fCache, zCache, sCache), p) { UUID.randomUUID().toString() }
        saveFolders(ns.folders)
        saveZikrs(ns.zikrs)
        val created = ns.folders.map { it.id }.filter { it !in before }.toSet()
        val hide = p.folders.filter { it.home != true }.mapNotNull { it.id }.filter { it in created }
        if (hide.isNotEmpty()) setHidden(hidden() + hide)      // only brand-new default folders start hidden
        return true
    }

    /** One time: the three fixed reminders become editable schedules (same on/off state). */
    private fun migrateSchedules() {
        if (sp.contains(K_SCHED)) return
        saveSchedules(
            listOf(
                Schedule("m", "Morning azkar", "Time for your morning azkar · tap to read", 6, 0, reminderOn("m", true), "morning"),
                Schedule("e", "Evening azkar", "Time for your evening azkar · tap to read", 17, 0, reminderOn("e", true), "evening"),
                Schedule("s", "Before sleep", "Time for your before sleep azkar · tap to read", 22, 0, reminderOn("s", false), "sleep")
            )
        )
    }

    // ---------- Parsing / saving (the only code that touches the caches) ----------
    private fun parseZikrs(raw: String): List<Zikr> {
        val arr = JSONArray(raw)
        return (0 until arr.length()).map {
            val o = arr.getJSONObject(it)
            Zikr(o.getString("id"), o.getString("ar"), o.optString("note", ""), o.optInt("n", 1), if (o.isNull("f")) null else o.getString("f"))
        }
    }

    private fun parseFolders(raw: String): List<Folder> {
        val arr = JSONArray(raw)
        return (0 until arr.length()).map {
            val o = arr.getJSONObject(it)
            Folder(
                o.getString("id"), o.getString("name"),
                if (o.isNull("ic")) null else o.optString("ic", "").ifEmpty { null },
                o.optBoolean("rm", false)
            )
        }
    }

    private fun parseSchedules(raw: String): List<Schedule> {
        val arr = JSONArray(raw)
        return (0 until arr.length()).map {
            val o = arr.getJSONObject(it)
            Schedule(
                o.getString("id"), o.optString("t", ""), o.optString("m", ""), o.optInt("h", 8), o.optInt("mi", 0),
                o.optBoolean("on", true), if (o.isNull("open")) null else o.getString("open")
            )
        }
    }

    @Synchronized
    private fun saveZikrs(list: List<Zikr>) {
        zCache = list.toList()
        val arr = JSONArray()
        list.forEach {
            arr.put(JSONObject().put("id", it.id).put("ar", it.arabic).put("note", it.note).put("n", it.repeat)
                .put("f", it.folderId ?: JSONObject.NULL))
        }
        sp.edit().putString(K_ZIKRS, arr.toString()).apply()
    }

    @Synchronized
    private fun saveFolders(list: List<Folder>) {
        fCache = list.toList()
        val arr = JSONArray()
        list.forEach { f ->
            val o = JSONObject().put("id", f.id).put("name", f.name)
            f.icon?.let { o.put("ic", it) }
            if (f.remember) o.put("rm", true)
            arr.put(o)
        }
        sp.edit().putString(K_FOLDERS, arr.toString()).apply()
    }

    @Synchronized
    private fun saveSchedules(list: List<Schedule>) {
        sCache = list.toList()
        val arr = JSONArray()
        list.forEach {
            arr.put(JSONObject().put("id", it.id).put("t", it.title).put("m", it.message).put("h", it.hour)
                .put("mi", it.minute).put("on", it.enabled).put("open", it.open ?: JSONObject.NULL))
        }
        sp.edit().putString(K_SCHED, arr.toString()).apply()
    }

    // ---------- Azkar ----------
    fun zikrs(): List<Zikr> = zCache
    fun zikr(id: String): Zikr? = zCache.firstOrNull { it.id == id }
    fun zikrsIn(folderId: String): List<Zikr> = zCache.filter { it.folderId == folderId }

    /** Number of azkar per folder id (null key = Anytime), computed in one pass. */
    fun zikrCounts(): Map<String?, Int> = zCache.groupingBy { it.folderId }.eachCount()

    fun addZikr(arabic: String, note: String, repeat: Int, folderId: String?): Zikr {
        zCache.firstOrNull { it.arabic == arabic && it.folderId == folderId }?.let {
            val merged = it.copy(note = note.ifBlank { it.note }, repeat = repeat)
            updateZikr(merged)
            return merged
        }
        val z = Zikr(UUID.randomUUID().toString(), arabic, note, repeat, folderId)
        saveZikrs(zCache + z)
        return z
    }

    fun updateZikr(z: Zikr) = saveZikrs(zCache.map { if (it.id == z.id) z else it })
    fun deleteZikr(id: String) {
        saveZikrs(zCache.filter { it.id != id })
        if (id in hzCache) setZikrsHidden(listOf(id), false)
    }

    /** Takes a zikr out of its folder (it stays saved as "Anytime"). */
    fun unfile(id: String) = saveZikrs(zCache.map { if (it.id == id) it.copy(folderId = null) else it })

    /** Reorders the zikrs of one folder; other azkar keep their relative position. */
    fun reorderFolder(folderId: String, orderedIds: List<String>) {
        val all = zCache
        val byId = all.associateBy { it.id }
        val queue = orderedIds.mapNotNull { byId[it] }.toMutableList()
        saveZikrs(all.map { if (it.folderId == folderId && queue.isNotEmpty()) queue.removeAt(0) else it })
    }

    // ---------- Folders ----------
    fun folders(): List<Folder> = fCache
    fun folder(id: String): Folder? = fCache.firstOrNull { it.id == id }

    fun addFolder(name: String, icon: String? = null): Folder {
        val f = Folder(UUID.randomUUID().toString(), name, icon?.takeUnless { EmojiRules.isRestricted(it) })
        saveFolders(fCache + f)
        return f
    }

    fun renameFolder(id: String, name: String) = saveFolders(fCache.map { if (it.id == id) it.copy(name = name) else it })
    fun setFolderIcon(id: String, icon: String?) = saveFolders(
        fCache.map { if (it.id == id) it.copy(icon = icon?.takeUnless { e -> EmojiRules.isRestricted(e) }) else it }
    )

    /** One time per start: folders that already use a people/animal emoji fall back to the default icon. */
    private fun sanitizeIcons() {
        if (fCache.any { it.icon != null && EmojiRules.isRestricted(it.icon) }) {
            saveFolders(fCache.map { if (it.icon != null && EmojiRules.isRestricted(it.icon)) it.copy(icon = null) else it })
        }
    }

    fun moveFolder(id: String, d: Int) {
        val l = fCache.toMutableList()
        val i = l.indexOfFirst { it.id == id }
        val j = i + d
        if (i < 0 || j !in l.indices) return
        val t = l[i]; l[i] = l[j]; l[j] = t
        saveFolders(l)
    }

    fun folderIndex(id: String) = fCache.indexOfFirst { it.id == id }

    /** Deletes the folder; its azkar stay saved as "Anytime" and reminders pointing at it just open the app. */
    fun deleteFolder(id: String) {
        saveFolders(fCache.filter { it.id != id })
        saveZikrs(zCache.map { if (it.folderId == id) it.copy(folderId = null) else it })
        if (sCache.any { it.open == id }) saveSchedules(sCache.map { if (it.open == id) it.copy(open = null) else it })
        setHomeOrder(homeOrder().filter { it != id })
        setHidden(hidden() - id)
        clearPosition(id)
    }

    fun folderName(key: String?): String {
        if (key == null) return "Anytime"
        return folder(key)?.name ?: Builtins.byKey(key)?.title ?: "Anytime"
    }

    fun isBuiltinFolder(key: String?) = Builtins.byKey(key) != null

    // ---------- Import / export ----------
    fun importAll(p: ImportJson.Parsed.Ok): ImportMerge.Summary {
        val (ns, sum) = ImportMerge.apply(ImportMerge.State(fCache, zCache, sCache), p) { UUID.randomUUID().toString() }
        saveFolders(ns.folders); saveZikrs(ns.zikrs); saveSchedules(ns.schedules)
        val sp2 = p.settings?.arabicSp ?: p.settings?.arabicSize?.let { ReaderPrefs.spFromOldIndex(it) }
        sp2?.let { setArabicSp(it) }
        p.settings?.notificationsOn?.let { setNotificationsOn(it) }
        return sum
    }

    fun exportJson(): String = ExportJson.build(ImportMerge.State(fCache, zCache, sCache), arabicSize(), notificationsOn(), arabicSp())

    // ---------- Home customisation ----------
    val defaultOrder: List<String> get() = fCache.map { it.id }

    fun homeOrder(): List<String> {
        val valid = defaultOrder
        val raw: String? = sp.getString(K_ORDER, null)
        val saved: List<String> = if (raw == null) emptyList() else {
            val arr = JSONArray(raw)
            (0 until arr.length()).map { arr.getString(it) }
        }
        return saved.filter { it in valid }.distinct().let { kept -> kept + valid.filter { it !in kept } }
    }

    fun setHomeOrder(order: List<String>) {
        sp.edit().putString(K_ORDER, JSONArray(order).toString()).apply()
    }

    fun hidden(): Set<String> {
        val arr = JSONArray(sp.getString(K_HIDDEN, "[]"))
        return (0 until arr.length()).map { arr.getString(it) }.toSet()
    }

    fun setHidden(set: Set<String>) {
        sp.edit().putString(K_HIDDEN, JSONArray(set.toList()).toString()).apply()
    }

    fun placement(): Placement = Placement.from(sp.getString(K_PLACE, null))
    fun setPlacement(p: Placement) = sp.edit().putString(K_PLACE, p.key).apply()

    /** Resets order/placement; default folders that are not meant for Home (no "home": true) are hidden again. */
    fun resetHome() {
        sp.edit().remove(K_ORDER).remove(K_HIDDEN).remove(K_PLACE).apply()
        val hide = defaults()?.folders?.filter { it.home != true }?.mapNotNull { it.id }.orEmpty()
        setHidden(hide.filter { folder(it) != null }.toSet())
    }

    // ---------- Settings ----------
    /** 0 = small, 1 = medium (default), 2 = large. */
    fun arabicSize() = sp.getInt(K_SIZE, 1)
    fun setArabicSize(v: Int) = sp.edit().putInt(K_SIZE, v).apply()

    // ---------- Per-folder reading position ----------
    fun rememberPosition(folderId: String) = folder(folderId)?.remember == true

    fun setFolderRemember(id: String, on: Boolean) {
        saveFolders(fCache.map { if (it.id == id) it.copy(remember = on) else it })
        if (!on) clearPosition(id)          // old progress must never leak back in
    }

    fun savePosition(folderId: String, zikrId: String, count: Int) {
        sp.edit().putString(K_POS + folderId, zikrId).putInt(K_POSC + folderId, count).apply()
    }

    fun savedPosition(folderId: String): Pair<String, Int>? {
        val id = sp.getString(K_POS + folderId, null) ?: return null
        return id to sp.getInt(K_POSC + folderId, 0)
    }

    fun clearPosition(folderId: String) {
        sp.edit().remove(K_POS + folderId).remove(K_POSC + folderId).apply()
    }

    // ---------- Reader settings ----------
    /** Arabic size in sp. Existing Small/Medium/Large choices carry over as 20/26/32. */
    fun arabicSp(): Int = ReaderPrefs.clampSp(sp.getInt(K_SP, ReaderPrefs.spFromOldIndex(arabicSize())))
    fun setArabicSp(v: Int) = sp.edit().putInt(K_SP, ReaderPrefs.clampSp(v)).apply()
    fun tapAdvance() = sp.getBoolean(K_TAP, true)
    fun setTapAdvance(v: Boolean) = sp.edit().putBoolean(K_TAP, v).apply()
    fun keepScreenOn() = sp.getBoolean(K_AWAKE, true)
    fun setKeepScreenOn(v: Boolean) = sp.edit().putBoolean(K_AWAKE, v).apply()

    /** 0 = current teal theme (default), 1 = OLED black. */
    fun theme() = sp.getInt(K_THEME, 0)
    fun setTheme(v: Int) = sp.edit().putInt(K_THEME, v).apply()

    private fun parseIds(raw: String): Set<String> {
        val a = JSONArray(raw)
        return (0 until a.length()).map { a.getString(it) }.toSet()
    }

    fun hiddenZikrs(): Set<String> = hzCache

    @Synchronized
    fun setZikrsHidden(ids: kotlin.collections.Collection<String>, hide: Boolean) {
        hzCache = Visibility.apply(hzCache, ids, hide)
        sp.edit().putString(K_HZ, JSONArray(hzCache.toList()).toString()).apply()
    }

    fun reminderOn(key: String, default: Boolean) = sp.getBoolean(K_REM + key, default)

    // ---------- Notification schedules ----------
    fun notificationsOn() = sp.getBoolean(K_NOTIF, true)
    fun setNotificationsOn(v: Boolean) = sp.edit().putBoolean(K_NOTIF, v).apply()

    fun schedules(): List<Schedule> = sCache
    fun schedule(id: String): Schedule? = sCache.firstOrNull { it.id == id }
    fun newScheduleId(): String = UUID.randomUUID().toString()

    fun upsertSchedule(s: Schedule) {
        saveSchedules(if (sCache.any { it.id == s.id }) sCache.map { if (it.id == s.id) s else it } else sCache + s)
    }

    fun deleteSchedule(id: String) = saveSchedules(sCache.filter { it.id != id })

    // ---------- Reader content ----------
    fun readerItems(key: String): List<ReaderItem> {
        if (key.startsWith("zikr:")) {
            return zikr(key.removePrefix("zikr:"))?.let { listOf(ReaderItem(it.arabic, it.repeat, it.id)) } ?: emptyList()
        }
        val src = if (key == "anytime") zCache.filter { it.folderId == null } else zikrsIn(key)
        return Visibility.visible(src, hzCache).map { ReaderItem(it.arabic, it.repeat, it.id) }
    }

    fun readerKey(z: Zikr): String = z.folderId ?: "anytime"

    fun readerIndex(z: Zikr): Int {
        val src = if (z.folderId == null) zCache.filter { it.folderId == null } else zikrsIn(z.folderId)
        return Visibility.readerIndex(src, hzCache, z.id).coerceAtLeast(0)
    }
}

package com.azkar.app.data

/** Pure rules for hiding azkar from the reading sequence. Hiding never changes a zikr itself. */
object Visibility {
    /** The azkar the reader shows, in their normal folder order. */
    fun visible(items: List<Zikr>, hidden: Set<String>): List<Zikr> = items.filter { it.id !in hidden }

    /** Reader position of [id] among the visible azkar, or -1 if it is hidden or missing. */
    fun readerIndex(items: List<Zikr>, hidden: Set<String>, id: String): Int =
        visible(items, hidden).indexOfFirst { it.id == id }

    fun apply(hidden: Set<String>, ids: kotlin.collections.Collection<String>, hide: Boolean): Set<String> =
        if (hide) hidden + ids else hidden - ids.toSet()
}

package com.azkar.app.reminders

import android.annotation.SuppressLint
import android.app.AlarmManager
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.net.Uri
import android.os.Build
import android.view.View
import android.widget.RemoteViews
import androidx.core.app.NotificationCompat
import androidx.core.app.NotificationManagerCompat
import androidx.core.content.ContextCompat
import com.azkar.app.MainActivity
import com.azkar.app.R
import com.azkar.app.ThemeMode
import com.azkar.app.data.Builtins
import com.azkar.app.data.Schedule
import com.azkar.app.data.Store
import java.util.Calendar

object ReminderScheduler {
    const val CHANNEL = "azkar_reminders_v2"
    private const val ACTION = "com.azkar.app.REMINDER"

    fun createChannel(ctx: Context) {
        val ch = NotificationChannel(CHANNEL, "Daily reminders", NotificationManager.IMPORTANCE_HIGH).apply {
            description = "Your azkar reminders"
            enableVibration(true)
            setShowBadge(true)
        }
        ctx.getSystemService(NotificationManager::class.java).createNotificationChannel(ch)
    }

    private fun alarms(ctx: Context) = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    /** Exact alarms need a user grant on Android 12+. */
    fun canExact(ctx: Context) = Build.VERSION.SDK_INT < 31 || alarms(ctx).canScheduleExactAlarms()

    // One alarm per schedule id (unique data URI), so editing replaces and deleting cancels exactly that alarm.
    private fun intentFor(ctx: Context, id: String) =
        Intent(ctx, ReminderReceiver::class.java).setAction(ACTION).setData(Uri.parse("azkar://reminder/$id")).putExtra("id", id)

    private fun pending(ctx: Context, id: String): PendingIntent =
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)

    /** Alarms from the old fixed reminders (request codes 0..2, no action). */
    private fun cancelLegacy(ctx: Context) {
        for (i in 0..2) {
            PendingIntent.getBroadcast(ctx, i, Intent(ctx, ReminderReceiver::class.java), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
                ?.let { alarms(ctx).cancel(it); it.cancel() }
        }
    }

    fun rescheduleAll(ctx: Context) {
        cancelLegacy(ctx)
        Store.schedules().forEach { apply(ctx, it) }
    }

    /** Schedules or cancels one reminder according to the global and individual switches. */
    fun apply(ctx: Context, s: Schedule) {
        if (Store.notificationsOn() && s.enabled) schedule(ctx, s) else cancel(ctx, s.id)
    }

    fun schedule(ctx: Context, s: Schedule) {
        val cal = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, s.hour); set(Calendar.MINUTE, s.minute)
            set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0)
            if (timeInMillis <= System.currentTimeMillis()) add(Calendar.DAY_OF_YEAR, 1)
        }
        val am = alarms(ctx)
        if (canExact(ctx)) am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, cal.timeInMillis, pending(ctx, s.id))
        else am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, cal.timeInMillis, pending(ctx, s.id))
    }

    fun cancel(ctx: Context, id: String) {
        PendingIntent.getBroadcast(ctx, 0, intentFor(ctx, id), PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE)
            ?.let { alarms(ctx).cancel(it); it.cancel() }
    }

    @SuppressLint("MissingPermission")
    fun notify(ctx: Context, s: Schedule) {
        if (Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(ctx, android.Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) return
        val nid = 100 + (s.id.hashCode() and 0xFFFF)
        val open = PendingIntent.getActivity(
            ctx, nid,
            Intent(ctx, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
                .apply { s.open?.let { putExtra("open", it) } },
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val ar = s.open?.let { Builtins.byKey(it)?.homeArabic } ?: ""
        // The system UI inflates RemoteViews with the device's night setting, not ours,
        // so each theme has its own layout built only from colours that never flip.
        val layout = if (Store.theme() == ThemeMode.OLED) R.layout.notification_reminder_oled else R.layout.notification_reminder
        val rv = RemoteViews(ctx.packageName, layout).apply {
            setTextViewText(R.id.n_title, s.title)
            setTextViewText(R.id.n_body, s.message)
            setTextViewText(R.id.n_ar, ar)
            setViewVisibility(R.id.n_ar, if (ar.isEmpty()) View.GONE else View.VISIBLE)
        }
        val n = NotificationCompat.Builder(ctx, CHANNEL)
            .setSmallIcon(R.drawable.ic_bell_1p75)
            .setColor(ctx.getColor(R.color.gold))
            .setContentTitle(s.title)
            .setContentText(s.message)
            .setCustomContentView(rv)
            .setCustomBigContentView(rv)
            .setCategory(NotificationCompat.CATEGORY_REMINDER)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
            .setContentIntent(open)
            .setAutoCancel(true)
            .build()
        NotificationManagerCompat.from(ctx).notify(nid, n)
    }
}

class ReminderReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        Store.init(ctx)
        ReminderScheduler.createChannel(ctx)
        val id = intent.getStringExtra("id") ?: return          // ignores old-style alarms
        val s = Store.schedule(id) ?: return                    // deleted schedule: nothing to show
        if (!Store.notificationsOn() || !s.enabled) return
        ReminderScheduler.notify(ctx, s)
        ReminderScheduler.schedule(ctx, s)                      // next day
    }
}

class BootReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        if (intent.action == Intent.ACTION_BOOT_COMPLETED) {
            Store.init(ctx)
            ReminderScheduler.createChannel(ctx)
            ReminderScheduler.rescheduleAll(ctx)
        }
    }
}

data class Card(
    val id: String, val name: String, val count: String, val emoji: String?, val vector: Int,
    val tint: Int, val isNew: Boolean, val onClick: () -> Unit

private val ARABIC = Regex("[\u0600-\u06FF]")

class CardAdapter : ListAdapter<Card, CardAdapter.VH>(object : DiffUtil.ItemCallback<Card>() {
    override fun areItemsTheSame(a: Card, b: Card) = a.id == b.id
    override fun areContentsTheSame(a: Card, b: Card) =
        a.name == b.name && a.count == b.count && a.emoji == b.emoji && a.vector == b.vector && a.tint == b.tint && a.isNew == b.isNew

        init {
            v.setOnClickListener { val p = bindingAdapterPosition; if (p != RecyclerView.NO_POSITION) getItem(p).onClick() }
        }

    override fun onCreateViewHolder(parent: ViewGroup, t: Int) =
        VH(LayoutInflater.from(parent.context).inflate(R.layout.item_folder_card, parent, false))

    override fun onBindViewHolder(h: VH, position: Int) {
        val c = getItem(position)
        val ctx = h.itemView.context

        h.count.text = c.count
        h.count.visibility = if (c.count.isEmpty()) View.GONE else View.VISIBLE
        // Views are recycled: every property is set in both branches.
        if (!c.isNew && ARABIC.containsMatchIn(c.name)) {
            h.name.typeface = h.amiri
            h.name.textSize = 22f
            h.name.textDirection = View.TEXT_DIRECTION_RTL

/** 12dp gaps between cards in a 2-column grid. */
class GridGap(private val dp: Int, private val span: Int = 2) : RecyclerView.ItemDecoration() {
    override fun getItemOffsets(out: Rect, v: View, parent: RecyclerView, s: RecyclerView.State) {
        val pos = parent.getChildAdapterPosition(v)
        if (pos == RecyclerView.NO_POSITION) return
        val px = (dp * v.resources.displayMetrics.density).toInt()
        val col = pos % span
        out.left = if (col == 0) 0 else px / 2
        out.right = if (col == 0) px / 2 else 0
        out.top = if (pos >= span) px else 0
    }
}

/** Lists a folder's azkar. Tap = read from there. Long-press or Select = hide or show several at once. */
class FolderFragment : Fragment(R.layout.fragment_folder_view) {
    private var selecting = false
    private val selected = LinkedHashSet<String>()

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val ctx = requireContext()
        val id = requireArguments().getString("id") ?: "anytime"
        if (id != "anytime" && Store.folder(id) == null) { nav().pop(); return }   // deleted from the editor

        view.findViewById<View>(R.id.btn_back).setOnClickListener { nav().pop() }
        view.findViewById<TextView>(R.id.title).text = Store.folderName(id)
        view.findViewById<View>(R.id.btn_edit).apply {
            if (id == "anytime") visibility = View.INVISIBLE
            else setOnClickListener { nav().push(EditFolderFragment.of(id)) }
        }

        val empty = view.findViewById<View>(R.id.empty)
        val info = view.findViewById<TextView>(R.id.hidden_info)
        val btnSelect = view.findViewById<TextView>(R.id.btn_select)
        val bar = view.findViewById<View>(R.id.action_bar)
        val selCount = view.findViewById<TextView>(R.id.sel_count)
        val btnAll = view.findViewById<View>(R.id.btn_all)
        val btnHide = view.findViewById<View>(R.id.btn_hide)
        val btnShow = view.findViewById<View>(R.id.btn_show)

        fun items() = if (id == "anytime") Store.zikrs().filter { it.folderId == null } else Store.zikrsIn(id)

        var refresh: () -> Unit = {}
        // While selecting, Back leaves selection mode instead of leaving the folder.
        val backCb = object : OnBackPressedCallback(false) {
            override fun handleOnBackPressed() { selecting = false; selected.clear(); refresh() }
        }
        requireActivity().onBackPressedDispatcher.addCallback(viewLifecycleOwner, backCb)

        val adapter = ZikrAdapter(
            onTap = { row ->
                if (selecting) { if (!selected.remove(row.id)) selected.add(row.id); refresh() }
                else if (row.hidden) Toast.makeText(ctx, "Hidden azkar are skipped while reading. Tap Select, then Show.", Toast.LENGTH_SHORT).show()
                else nav().push(ReaderFragment.of(id, Visibility.readerIndex(items(), Store.hiddenZikrs(), row.id).coerceAtLeast(0)))
            },
            onLong = { row -> if (!selecting) { selecting = true; selected.add(row.id); refresh() } },
            onEdit = { zid -> nav().push(EditZikrFragment.of(zid)) }
        )

        refresh = {
            val hidden = Store.hiddenZikrs()
            val list = items()
            selected.retainAll(list.map { it.id }.toSet())
            val nHidden = list.count { it.id in hidden }
            empty.visibility = if (list.isEmpty()) View.VISIBLE else View.GONE
            info.text = when (nHidden) { 0 -> ""; 1 -> "1 hidden"; else -> "$nHidden hidden" }
            btnSelect.text = if (selecting) "Cancel" else "Select"
            btnSelect.visibility = if (list.isEmpty()) View.INVISIBLE else View.VISIBLE
            bar.visibility = if (selecting) View.VISIBLE else View.GONE
            selCount.text = "${selected.size} selected"
            btnHide.alpha = if (selected.isEmpty()) 0.4f else 1f
            btnShow.alpha = if (selected.isEmpty()) 0.4f else 1f
            backCb.isEnabled = selecting
            adapter.submitList(list.map { ZRow(it.id, it.arabic, it.repeat, it.note, it.id in hidden, selecting, it.id in selected) })
        }

        btnSelect.setOnClickListener { selecting = !selecting; selected.clear(); refresh() }
        btnAll.setOnClickListener {
            val ids = items().map { it.id }
            val all = selected.size == ids.size
            selected.clear()
            if (!all) selected.addAll(ids)
            refresh()
        }
        btnHide.setOnClickListener {
            if (selected.isNotEmpty()) { Store.setZikrsHidden(selected.toList(), true); selecting = false; selected.clear(); refresh() }
        }
        btnShow.setOnClickListener {
            if (selected.isNotEmpty()) { Store.setZikrsHidden(selected.toList(), false); selecting = false; selected.clear(); refresh() }
        }

        view.findViewById<RecyclerView>(R.id.list).apply {
            layoutManager = LinearLayoutManager(ctx)
            this.adapter = adapter
            setHasFixedSize(true)
            addItemDecoration(Gap(8))
            itemAnimator = null
        }
        refresh()
    }

    companion object {
        fun of(id: String) = FolderFragment().apply { arguments = bundleOf("id" to id) }

    private fun readAndConfirm(uri: Uri) {
        val ctx = requireContext()

            ctx.contentResolver.openInputStream(uri)?.use {

        } catch (e: Exception) {
            null
        }

        when {
            raw == null -> showError("The file could not be read.")

            raw.length > 5_000_000 ->
                showError("The file is too large (over 5 MB).")

            else -> {
                when (val p = ImportJson.parse(raw)) {
                    is ImportJson.Parsed.Error -> showError(p.message)
                    is ImportJson.Parsed.Ok -> confirm(p)

            }
        }

    private fun confirm(p: ImportJson.Parsed.Ok) {
        val nF = p.folders.size
        val nA = p.folders.sumOf { it.items.size } + p.unfiled.size

        val existing = p.folders.count { f ->
            Store.folders().any {
                it.id == f.id ||
                    it.name.trim().equals(f.name.trim(), ignoreCase = true)
            }
        }

        val lines = p.folders.take(8).joinToString("\n") {

package com.azkar.app.ui

/** Pure navigation rules of the reader (no Android classes), so they can be unit tested. */
class ReaderNav(private val targets: List<Int>, start: Int = 0, count: Int = 0) {
    enum class Outcome { MOVED, FINISH, BOUNCE }

    var index: Int = start.coerceIn(0, (targets.size - 1).coerceAtLeast(0)); private set
    var count: Int = count; private set

    val last get() = targets.size - 1
    val isLast get() = index == last
    /** The repetition count of the CURRENT zikr is complete. */
    val isDone get() = targets[index] <= 1 || count >= targets[index]

    /** Counter tap. Returns true if the count changed. */
    fun tapRing(): Boolean { if (isDone) return false; count++; return true }

    fun canPage(dir: Int) = if (dir > 0) index < last else index > 0
    /** Leaving the reader needs BOTH: last zikr of the collection AND its own count complete. */
    fun canFinish() = isLast && isDone

    /** Tap on the card or swipe left. */
    fun next(): Outcome = when {
        canPage(1) -> { index++; count = 0; Outcome.MOVED }
        canFinish() -> Outcome.FINISH
        else -> Outcome.BOUNCE
    }

    /** Swipe right. Never finishes. */
    fun prev(): Outcome =
        if (canPage(-1)) { index--; count = 0; Outcome.MOVED } else Outcome.BOUNCE
}

    fun px(ctx: Context, dp: Float): Float = dp * ctx.resources.displayMetrics.density
    fun pxi(ctx: Context, dp: Float): Int = px(ctx, dp).toInt()
    fun color(ctx: Context, @ColorRes id: Int) = ctx.getColor(id)

    fun shape(ctx: Context, fill: Int?, stroke: Int?, radiusDp: Float, dashed: Boolean = false): GradientDrawable =
        GradientDrawable().apply {

            cornerRadius = px(ctx, radiusDp)

            if (stroke != null) {
                if (dashed) setStroke(pxi(ctx, 1f), stroke, px(ctx, 3f), px(ctx, 3f))
                else setStroke(pxi(ctx, 1f), stroke)
            }
        }

        LinearLayout.LayoutParams(w, h).also { if (ctx != null && top != 0) it.topMargin = pxi(ctx, top.toFloat()) }

    /** Rows of [cols] equally wide cells with [gap]dp between them. */
    fun grid(ctx: Context, cells: List<View>, cols: Int, gap: Int): LinearLayout {
        val root = LinearLayout(ctx).apply { orientation = LinearLayout.VERTICAL }
        cells.chunked(cols).forEachIndexed { r, rowCells ->
            val row = LinearLayout(ctx).apply { orientation = LinearLayout.HORIZONTAL }
            for (c in 0 until cols) {
                val cell = rowCells.getOrNull(c) ?: View(ctx)
                val p = LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f)
                if (c > 0) p.marginStart = pxi(ctx, gap.toFloat())
                row.addView(cell, p)
            }
            root.addView(row, LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.WRAP_CONTENT).also {
                if (r > 0) it.topMargin = pxi(ctx, gap.toFloat())
            })
        }
        return root
    }

            else shape(ctx, null, ctx.getColor(R.color.border2), 22f)

        }

/** Card-follows-finger swipe layout. dir: +1 = next (swipe left), -1 = previous (swipe right). */
class SwipeLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : LinearLayout(ctx, attrs) {
    var card: View? = null
    var canPage: ((Int) -> Boolean)? = null
    var onPage: ((Int) -> Unit)? = null
    var tapEnabled = true
    var canFinish: (() -> Boolean)? = null
    var onFinish: (() -> Unit)? = null
    private var rawDx = 0f
    private var finishing = false

    private val vc = android.view.ViewConfiguration.get(ctx)
    private val slop = vc.scaledTouchSlop
    private val minFling = vc.scaledMinimumFlingVelocity * 8f
    private val dp = resources.displayMetrics.density
    private var downX = 0f
    private var downY = 0f
    private var dragging = false
    private var tapOk = false
    private var busy = false
    private var vt: android.view.VelocityTracker? = null

    /** Stops the system back gesture from stealing swipes that start at the screen edge. */
    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        super.onLayout(changed, l, t, r, b)
        if (android.os.Build.VERSION.SDK_INT >= 29) {
            val w = r - l
            val h = b - t
            val c = card
            val cy = if (c != null) (c.top + c.bottom) / 2 else h / 2
            val half = (100 * dp).toInt()
            val top = (cy - half).coerceAtLeast(0)
            val bottom = (cy + half).coerceAtMost(h)
            val edge = (40 * dp).toInt()
            setSystemGestureExclusionRects(
                listOf(android.graphics.Rect(0, top, edge, bottom), android.graphics.Rect(w - edge, top, w, bottom))
            )
        }
    }

    private fun inCard(x: Float, y: Float): Boolean {
        val c = card ?: return false
        return x >= c.left && x <= c.right && y >= c.top && y <= c.bottom
    }

    private fun far(e: MotionEvent) =
        kotlin.math.hypot(e.x - downX, e.y - downY) > slop

    private fun horizontal(e: MotionEvent) =
        kotlin.math.abs(e.x - downX) > slop && kotlin.math.abs(e.x - downX) > kotlin.math.abs(e.y - downY) * 1.2f

    private fun start(e: MotionEvent) {
        downX = e.x; downY = e.y; dragging = false
        tapOk = inCard(e.x, e.y)
        vt?.recycle(); vt = android.view.VelocityTracker.obtain(); vt?.addMovement(e)
    }

    private fun finishTap(e: MotionEvent) {
        if (tapEnabled && tapOk && e.eventTime - e.downTime < android.view.ViewConfiguration.getLongPressTimeout()) next()
        tapOk = false
    }

    private fun cleanup() { vt?.recycle(); vt = null }

    override fun onInterceptTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
            }
            MotionEvent.ACTION_UP -> { if (!dragging) finishTap(e); cleanup() }
            MotionEvent.ACTION_CANCEL -> { tapOk = false; cleanup() }
        }
        return dragging
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        if (vt == null) vt = android.view.VelocityTracker.obtain()
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
                if (dragging) drag(e.x - downX)
            }
            MotionEvent.ACTION_UP -> {
                vt?.addMovement(e)
                if (dragging) release(true) else finishTap(e)
                dragging = false; cleanup()
            }
            MotionEvent.ACTION_CANCEL -> {
                if (dragging) release(false)
                dragging = false; tapOk = false; cleanup()
            }
        }
        return true
    }

    private fun dirOf(t: Float) = if (t < 0f) 1 else -1   // finger moved left -> next

    private fun drag(dx: Float) {
        val c = card ?: return
        rawDx = dx
        val d = dx - kotlin.math.sign(dx) * slop
        val ok = canPage?.invoke(dirOf(d)) ?: false
        val t = if (ok) d else d * 0.25f          // rubber band at the first/last azkar
        c.translationX = t
        c.alpha = 1f - 0.6f * minOf(kotlin.math.abs(t) / width.coerceAtLeast(1), 1f)
    }

    private fun release(up: Boolean) {
        val c = card ?: return
        vt?.computeCurrentVelocity(1000)
        val vx = vt?.xVelocity ?: 0f
        val t = c.translationX
        val dir = dirOf(t)
        val ok = canPage?.invoke(dir) ?: false
        val farEnough = kotlin.math.abs(t) > width * 0.28f
        val flung = kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(t) && kotlin.math.abs(t) > slop * 2
        val rawFar = kotlin.math.abs(rawDx) > 48 * dp ||
            (kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(rawDx))
        val finish = up && dir == 1 && !ok && rawFar && (canFinish?.invoke() == true)
        rawDx = 0f
        if (up && ok && (farEnough || flung)) commit(c, dir)
        else if (finish) { c.animate().cancel(); c.translationX = 0f; c.alpha = 1f; fireFinish() }
        else c.animate().translationX(0f).alpha(1f).setDuration(200)
            .setInterpolator(android.view.animation.DecelerateInterpolator()).start()
    }

    private fun fireFinish() {
        if (finishing) return
        finishing = true
        post { onFinish?.invoke() }   // never change fragments in the middle of touch dispatch
    }

    /** Next zikr; on the last one, leaves only if its count is complete, otherwise a small bounce. */
    fun next() {
        val c = card ?: return
        if (busy || finishing) return
        if (canPage?.invoke(1) == true) commit(c, 1)
        else if (canFinish?.invoke() == true) fireFinish()
        else c.animate().translationX(-12 * dp).setDuration(90)
            .withEndAction { c.animate().translationX(0f).setDuration(160).start() }.start()
    }

    private fun commit(c: View, dir: Int) {
        busy = true
        val w = width.toFloat()
        c.animate().translationX(-dir * w).alpha(0f).setDuration(140)
            .setInterpolator(android.view.animation.AccelerateInterpolator())
            .withEndAction {
                onPage?.invoke(dir)
                c.translationX = dir * w * 0.35f   // new card enters from the side you swiped toward
                c.alpha = 0f
                c.animate().translationX(0f).alpha(1f).setDuration(220)
                    .setInterpolator(android.view.animation.DecelerateInterpolator())
                    .withEndAction { busy = false }.start()
            }.start()
    }
}

/** Wrapping row layout with 8dp gaps (flex-wrap). */
class FlowLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : ViewGroup(ctx, attrs) {
    private val gap = (8 * resources.displayMetrics.density).toInt()

    override fun onMeasure(wSpec: Int, hSpec: Int) {
        val maxW = MeasureSpec.getSize(wSpec) - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            c.measure(MeasureSpec.makeMeasureSpec(maxW, MeasureSpec.AT_MOST), MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED))
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
        setMeasuredDimension(MeasureSpec.getSize(wSpec), y + lineH + paddingTop + paddingBottom)
    }

    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val maxW = r - l - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            c.layout(paddingLeft + x, paddingTop + y, paddingLeft + x + c.measuredWidth, paddingTop + y + c.measuredHeight)
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
    }
}

package com.azkar.app.ui

import android.text.Editable
import android.text.TextWatcher
import android.view.View
import android.view.inputmethod.EditorInfo
import android.widget.EditText
import com.azkar.app.R
import com.azkar.app.data.Store

/**
 * Shared behaviour of the Add zikr and Edit zikr screens:
 * repeat stepper and folder chips.
 */
class ZikrForm(
    private val view: View,
    private val allowNew: Boolean,
    private val onSelChanged: (String?) -> Unit = {}
) {
    /** Last valid repeat count. */
    private var count = 1

    /** Prevents TextWatcher recursion during programmatic updates. */
    private var syncing = false

    var n: Int
        get() = box.text.toString()
            .toIntOrNull()
            ?.takeIf { it >= 1 }
            ?.coerceAtMost(999) ?: count
        set(v) {
            count = v.coerceIn(1, 999)
            syncBox()
        }

    var sel: String? = null
        set(v) {
            field = v
            renderChips()
            onSelChanged(v)
        }

    val arabic: EditText = view.findViewById(R.id.ar)
    private val box: EditText = view.findViewById(R.id.n)

    init {
        view.findViewById<View>(R.id.btn_dec).setOnClickListener {
            n = n - 1
        }

        view.findViewById<View>(R.id.btn_inc).setOnClickListener {
            n = n + 1
        }

        box.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(
                s: CharSequence?, start: Int, count: Int, after: Int
            ) {}

            override fun onTextChanged(
                s: CharSequence?, start: Int, before: Int, count: Int
            ) {}

            override fun afterTextChanged(s: Editable?) {
                if (syncing) return

                s?.toString()
                    ?.toIntOrNull()
                    ?.takeIf { it >= 1 }
                    ?.let { count = it.coerceAtMost(999) }

                updateDec()
            }
        })

        box.setOnFocusChangeListener { _, focused ->
            if (!focused) syncBox()
        }

        box.setOnEditorActionListener { _, action, _ ->
            if (action == EditorInfo.IME_ACTION_DONE) {
                syncBox()
            }
            false
        }

        syncBox()
    }

    private fun syncBox() {
        syncing = true
        try {
            val value = count.toString()
            if (box.text.toString() != value) {
                box.setText(value)
            }
        } finally {
            syncing = false
        }

        updateDec()
    }

    private fun updateDec() {
        view.findViewById<View>(R.id.btn_dec).isEnabled = n > 1
    }

    fun renderChips() {
        val ctx = view.context
        val chips = view.findViewById<FlowLayout>(R.id.chips)

        chips.removeAllViews()

        val choices = Store.folders().map {
            it.id to ((it.icon?.let { e -> "$e " } ?: "") + it.name)
        }

        choices.forEach { (k, label) ->
            chips.addView(
                Ui.chip(ctx, label, sel == k, false) {
                    sel = k
                }
            )
        }

        if (allowNew) {
            chips.addView(
                Ui.chip(ctx, "+ New folder", sel == "new", true) {
                    sel = "new"
                }
            )
        }
    }
}

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        currentTab = savedInstanceState?.getInt("tab")?.let { Tab.values().getOrNull(it) } ?: Tab.HOME
        Store.init(this)
        if (Build.VERSION.SDK_INT >= 28) {
            window.attributes = window.attributes.apply {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
        }
        if (Build.VERSION.SDK_INT >= 33 &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) notifPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
        WindowCompat.setDecorFitsSystemWindows(window, false)
        WindowCompat.getInsetsController(window, window.decorView).apply {
            isAppearanceLightStatusBars = false
            isAppearanceLightNavigationBars = false
        }
        setContentView(R.layout.activity_main)
        if (savedInstanceState == null) {
            showRoot(HomeFragment())
            handleIntent(intent)
        }
    }

    override fun onSaveInstanceState(outState: Bundle) {
        super.onSaveInstanceState(outState)
        outState.putInt("tab", currentTab.ordinal)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        handleIntent(intent)
    }

    private fun handleIntent(i: Intent?) {
        val key = i?.getStringExtra("open") ?: return
        i.removeExtra("open")
        showRoot(HomeFragment())
        if (Store.readerItems(key).isNotEmpty()) push(ReaderFragment.of(key))   // deleted or empty folder: just stay on Home
    }

    private fun showRoot(f: Fragment) {
        supportFragmentManager.popBackStackImmediate(null, FragmentManager.POP_BACK_STACK_INCLUSIVE)
        supportFragmentManager.beginTransaction().replace(R.id.container, f).commit()
    }

    fun openTab(tab: Tab) {
        if (tab == currentTab && supportFragmentManager.backStackEntryCount == 0) return
        currentTab = tab
        showRoot(
            when (tab) {
                Tab.HOME -> HomeFragment()
                Tab.LIBRARY -> LibraryFragment()
                Tab.MY -> MyAzkarFragment()
            }
        )
    }

    fun push(f: Fragment) {
        supportFragmentManager.beginTransaction().replace(R.id.container, f).addToBackStack(null).commit()
    }

    fun pop() {
        android.util.Log.d("AzkarNav", "pop() called from:", Throwable())
        supportFragmentManager.popBackStack()
    }

    /** The last zikr of a collection was finished: back to Home. */
    fun finishReading() {
        currentTab = Tab.HOME
        showRoot(HomeFragment())
    }
    /** Resets the back stack and shows Home. */
    fun goHome() {
        android.util.Log.d("AzkarNav", "goHome() called from:", Throwable())
        currentTab = Tab.HOME
        showRoot(HomeFragment())
    }
}

import androidx.appcompat.app.AppCompatDelegate

/** The two in-app themes. Never follows the phone's system dark mode. */
object ThemeMode {
    const val CURRENT = 0
    const val OLED = 1

    /** OLED = night resources (values-night); Current = day resources. Recreates the activity if the mode changed. */
    fun apply(choice: Int) = AppCompatDelegate.setDefaultNightMode(
        if (choice == OLED) AppCompatDelegate.MODE_NIGHT_YES else AppCompatDelegate.MODE_NIGHT_NO
    )
}

package com.azkar.app.data

import android.content.Context
import android.content.SharedPreferences
import org.json.JSONArray
import org.json.JSONObject
import java.util.UUID

/** Persistent local storage (SharedPreferences + JSON). Survives app restarts. */
object Store {
    private lateinit var sp: SharedPreferences

    private const val K_ZIKRS = "zikrs"
    private const val K_FOLDERS = "folders"
    private const val K_ORDER = "home_order"
    private const val K_HIDDEN = "home_hidden"
    private const val K_PLACE = "home_place"
    private const val K_TRANS = "show_translation"
    private const val K_SIZE = "arabic_size"
    private const val K_REM = "reminder_"

    fun init(context: Context) {
        if (::sp.isInitialized) return
        sp = context.applicationContext.getSharedPreferences("azkar", Context.MODE_PRIVATE)
        if (!sp.contains(K_ZIKRS)) seed()
    }

    private fun seed() {
        saveFolders(listOf(Folder("family", "Family"), Folder("friday", "Friday")))
        saveZikrs(
            listOf(
                Zikr("seed1", "أَسْتَغْفِرُ اللَّهَ", "I seek forgiveness from Allah.", 100, null),
                Zikr("seed2", "لَا حَوْلَ وَلَا قُوَّةَ إِلَّا بِاللَّهِ", "There is no power and no strength except with Allah.", 10, "family"),
                Zikr("seed3", "اللَّهُمَّ صَلِّ وَسَلِّمْ عَلَى نَبِيِّنَا مُحَمَّدٍ", "O Allah, send prayers and peace upon our Prophet Muhammad.", 10, "friday")
            )
        )
    }

    // ---------- Azkar ----------
    fun zikrs(): List<Zikr> {
        val arr = JSONArray(sp.getString(K_ZIKRS, "[]"))
        return (0 until arr.length()).map {
            val o = arr.getJSONObject(it)
            Zikr(
                o.getString("id"), o.getString("ar"), o.optString("note", ""), o.optInt("n", 1),
                if (o.isNull("f")) null else o.getString("f")
            )
        }
    }

    private fun saveZikrs(list: List<Zikr>) {
        val arr = JSONArray()
        list.forEach {
            arr.put(JSONObject().put("id", it.id).put("ar", it.arabic).put("note", it.note).put("n", it.repeat)
                .put("f", it.folderId ?: JSONObject.NULL))
        }
        sp.edit().putString(K_ZIKRS, arr.toString()).apply()
    }

    fun zikr(id: String): Zikr? = zikrs().firstOrNull { it.id == id }
    fun zikrsIn(folderId: String): List<Zikr> = zikrs().filter { it.folderId == folderId }

    fun addZikr(arabic: String, note: String, repeat: Int, folderId: String?): Zikr {
        zikrs().firstOrNull { it.arabic == arabic && it.folderId == folderId }?.let {
            val merged = it.copy(note = note.ifBlank { it.note }, repeat = repeat)
            updateZikr(merged)
            return merged
        }
        val z = Zikr(UUID.randomUUID().toString(), arabic, note, repeat, folderId)
        saveZikrs(zikrs() + z)
        return z
    }

    fun updateZikr(z: Zikr) = saveZikrs(zikrs().map { if (it.id == z.id) z else it })
    fun deleteZikr(id: String) = saveZikrs(zikrs().filter { it.id != id })

    /** Takes a zikr out of its folder (it stays saved as "Anytime"). */
    fun unfile(id: String) = saveZikrs(zikrs().map { if (it.id == id) it.copy(folderId = null) else it })

    /** Reorders the zikrs of one folder; other azkar keep their relative position. */
    fun reorderFolder(folderId: String, orderedIds: List<String>) {
        val all = zikrs()
        val byId = all.associateBy { it.id }
        val queue = orderedIds.mapNotNull { byId[it] }.toMutableList()
        saveZikrs(all.map { if (it.folderId == folderId && queue.isNotEmpty()) queue.removeAt(0) else it })
    }

    // ---------- Folders ----------
    fun folders(): List<Folder> {
        val arr = JSONArray(sp.getString(K_FOLDERS, "[]"))
        return (0 until arr.length()).map { Folder(arr.getJSONObject(it).getString("id"), arr.getJSONObject(it).getString("name")) }
    }

    private fun saveFolders(list: List<Folder>) {
        val arr = JSONArray()
        list.forEach { arr.put(JSONObject().put("id", it.id).put("name", it.name)) }
        sp.edit().putString(K_FOLDERS, arr.toString()).apply()
    }

    fun folder(id: String): Folder? = folders().firstOrNull { it.id == id }

    fun addFolder(name: String): Folder {
        val f = Folder(UUID.randomUUID().toString(), name)
        saveFolders(folders() + f)
        return f
    }

    fun renameFolder(id: String, name: String) = saveFolders(folders().map { if (it.id == id) it.copy(name = name) else it })

    /** Deletes the folder; its azkar stay saved as "Anytime". */
    fun deleteFolder(id: String) {
        saveFolders(folders().filter { it.id != id })
        saveZikrs(zikrs().map { if (it.folderId == id) it.copy(folderId = null) else it })
        setHomeOrder(homeOrder().filter { it != id })
        setHidden(hidden() - id)
    }

    /** Display name for a folder key (built-in or custom). */
    fun folderName(key: String?): String {
        if (key == null) return "Anytime"
        Builtins.byKey(key)?.let { return it.title }
        return folder(key)?.name ?: "Anytime"
    }

    fun isBuiltinFolder(key: String?) = Builtins.byKey(key) != null

    // ---------- Home customisation ----------
    val defaultOrder: List<String> get() = Builtins.homeCollections.map { it.key } + folders().map { it.id }

    /** Effective order: saved order (pruned) followed by anything new. */
    fun homeOrder(): List<String> {
        val valid = defaultOrder
        val raw: String? = sp.getString(K_ORDER, null)
        val saved: List<String> = if (raw == null) emptyList() else {
            val arr = JSONArray(raw)
            (0 until arr.length()).map { arr.getString(it) }
        }
        return saved.filter { it in valid }.distinct().let { kept -> kept + valid.filter { it !in kept } }
    }

    fun setHomeOrder(order: List<String>) {
        sp.edit().putString(K_ORDER, JSONArray(order).toString()).apply()
    }

    fun hidden(): Set<String> {
        val arr = JSONArray(sp.getString(K_HIDDEN, "[]"))
        return (0 until arr.length()).map { arr.getString(it) }.toSet()
    }

    fun setHidden(set: Set<String>) {
        sp.edit().putString(K_HIDDEN, JSONArray(set.toList()).toString()).apply()
    }

    fun placement(): Placement = Placement.from(sp.getString(K_PLACE, null))
    fun setPlacement(p: Placement) = sp.edit().putString(K_PLACE, p.key).apply()

    fun resetHome() {
        sp.edit().remove(K_ORDER).remove(K_HIDDEN).remove(K_PLACE).apply()
    }

    // ---------- Reader / reminder settings ----------
    fun showTranslation() = sp.getBoolean(K_TRANS, true)
    fun setShowTranslation(v: Boolean) = sp.edit().putBoolean(K_TRANS, v).apply()

    /** 0 = small, 1 = medium (default), 2 = large. */
    fun arabicSize() = sp.getInt(K_SIZE, 1)
    fun setArabicSize(v: Int) = sp.edit().putInt(K_SIZE, v).apply()

    fun reminderOn(key: String, default: Boolean) = sp.getBoolean(K_REM + key, default)
    fun setReminderOn(key: String, v: Boolean) = sp.edit().putBoolean(K_REM + key, v).apply()

    // ---------- Reader content ----------
    /** Built-in azkar for a collection followed by azkar the user filed in it. */
    fun readerItems(key: String): List<ReaderItem> {
        if (key.startsWith("zikr:")) {
            return zikr(key.removePrefix("zikr:"))?.let { listOf(ReaderItem(it.arabic, it.repeat)) } ?: emptyList()
        }
        val base = Builtins.byKey(key)?.items ?: emptyList()
        return base + zikrsIn(key).map { ReaderItem(it.arabic, it.repeat) }
    }

    /** Reader key that opens [z] inside its own collection/folder ("Anytime" azkar open on their own). */
    fun readerKey(z: Zikr): String = z.folderId ?: "zikr:${z.id}"

    /** Position of [z] inside that reader list. */
    fun readerIndex(z: Zikr): Int {
        val f = z.folderId ?: return 0
        val base = Builtins.byKey(f)?.items?.size ?: 0
        return base + zikrsIn(f).indexOfFirst { it.id == z.id }.coerceAtLeast(0)
    }

    /** All saved azkar for the Library, grouped by category: built-ins, then your folders, then Anytime. */
    fun libraryZikrs(): List<Zikr> {
        val all = zikrs()
        val keys = ArrayList<String?>().apply {
            addAll(Builtins.library.map { it.key }); addAll(folders().map { it.id }); add(null)
        }
        return keys.flatMap { k -> all.filter { it.folderId == k } } + all.filter { it.folderId !in keys }
    }
}

class MainActivity : AppCompatActivity() {
    private var currentTab: Tab = Tab.HOME
    private val notifPermission = registerForActivityResult(ActivityResultContracts.RequestPermission()) { }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        Store.init(this)
        if (Build.VERSION.SDK_INT >= 28) {
            window.attributes = window.attributes.apply {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
        }
        if (Build.VERSION.SDK_INT >= 33 &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) notifPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
        WindowCompat.setDecorFitsSystemWindows(window, false)
        WindowCompat.getInsetsController(window, window.decorView).apply {
            isAppearanceLightStatusBars = false
            isAppearanceLightNavigationBars = false
        }
        setContentView(R.layout.activity_main)
        if (savedInstanceState == null) {
            showRoot(HomeFragment())
            handleIntent(intent)
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        handleIntent(intent)
    }

    /** A reminder notification opens its collection in the reader. */
    private fun handleIntent(i: Intent?) {
        val key = i?.getStringExtra("open") ?: return
        i.removeExtra("open")
        showRoot(HomeFragment())
        push(ReaderFragment.of(key))
    }

    private fun showRoot(f: Fragment) {
        supportFragmentManager.popBackStackImmediate(null, FragmentManager.POP_BACK_STACK_INCLUSIVE)
        supportFragmentManager.beginTransaction().replace(R.id.container, f).commit()
    }

    fun openTab(tab: Tab) {
        if (tab == currentTab && supportFragmentManager.backStackEntryCount == 0) return
        currentTab = tab
        showRoot(
            when (tab) {
                Tab.HOME -> HomeFragment()
                Tab.LIBRARY -> LibraryFragment()
                Tab.MY -> MyAzkarFragment()
                Tab.REMINDERS -> RemindersFragment()
            }
        )
    }

    fun push(f: Fragment) {
        supportFragmentManager.beginTransaction().replace(R.id.container, f).addToBackStack(null).commit()
    }

    fun pop() {
        supportFragmentManager.popBackStack()
    }
}

class ReaderFragment : Fragment(R.layout.fragment_reader) {
    private var i = 0
    private var count = 0
    private var lastBack = 0L

    override fun onSaveInstanceState(outState: Bundle) {
        super.onSaveInstanceState(outState)
        outState.putInt("i", i); outState.putInt("count", count)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val ctx = requireContext()
        view.keepScreenOn = true

        val key = requireArguments().getString("key") ?: "morning"
        val col = Builtins.byKey(key)
        val items = Store.readerItems(key)
        if (items.isEmpty()) { nav().pop(); return }
        i = (savedInstanceState?.getInt("i") ?: requireArguments().getInt("start", 0)).coerceIn(0, items.size - 1)
        count = savedInstanceState?.getInt("count") ?: 0
        val title = when {
            key in listOf("morning", "evening", "afterPrayer", "sleep", "waking") -> (col?.title ?: "") + " azkar"
            col != null -> col.title
            else -> Store.folderName(key)
        }
        val sizeSp = intArrayOf(20, 26, 32)[Store.arabicSize().coerceIn(0, 2)]
        val last = items.size - 1
        val gold = ctx.getColor(R.color.gold)
        val teal = ctx.getColor(R.color.teal)
        val track = ctx.getColor(R.color.track)

        // Full screen: small margins plus whatever the cutout / bars need.
        val d = resources.displayMetrics.density
        ViewCompat.setOnApplyWindowInsetsListener(view) { v, insets ->
            val b = insets.getInsets(WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout())
            v.setPadding((16 * d).toInt() + b.left, (12 * d).toInt() + b.top, (16 * d).toInt() + b.right, (16 * d).toInt() + b.bottom)
            insets
        }
        ViewCompat.requestApplyInsets(view)

        // A stray system back (edge swipe) must not close the reader: it needs a second press.
        requireActivity().onBackPressedDispatcher.addCallback(viewLifecycleOwner, object : OnBackPressedCallback(true) {
            override fun handleOnBackPressed() {
                val now = SystemClock.uptimeMillis()
                if (now - lastBack < 2000) nav().pop()
                else { lastBack = now; Toast.makeText(ctx, "Press back again to leave", Toast.LENGTH_SHORT).show() }
            }
        })

        view.findViewById<View>(R.id.btn_back).setOnClickListener { nav().pop() }
        view.findViewById<TextView>(R.id.title).text = title
        val ar = view.findViewById<TextView>(R.id.ar).apply {
            textSize = sizeSp.toFloat()
            val lhPx = android.util.TypedValue.applyDimension(android.util.TypedValue.COMPLEX_UNIT_SP, sizeSp * 1.85f, resources.displayMetrics)
            androidx.core.widget.TextViewCompat.setLineHeight(this, lhPx.toInt())
        }
        val arScroll = view.findViewById<ScrollView>(R.id.ar_scroll)
        val segs = view.findViewById<LinearLayout>(R.id.segs)
        val ring = view.findViewById<RingView>(R.id.ring)
        val countTv = view.findViewById<TextView>(R.id.count)
        val sub = view.findViewById<TextView>(R.id.sub)
        val hint = view.findViewById<TextView>(R.id.hint)
        val pos = view.findViewById<TextView>(R.id.pos)

        fun render() {
            val cur = items[i]
            val done = count >= cur.target
            ar.text = cur.arabic
            pos.text = "${i + 1} of ${items.size}"
            segs.removeAllViews()
            items.indices.forEach { k ->
                val c = if (k < i) teal else if (k == i) gold else track
                val seg = View(ctx).apply { background = Ui.shape(ctx, c, null, 2f) }
                val p = LinearLayout.LayoutParams(0, Ui.pxi(ctx, 4f), 1f)
                if (k > 0) p.marginStart = Ui.pxi(ctx, 6f)
                segs.addView(seg, p)
            }
            countTv.text = count.toString()
            sub.text = if (done) "Done" else "of ${cur.target}"
            ring.setProgress(Math.round(count.toFloat() / cur.target * 360f).toFloat(), if (done) teal else gold)
            hint.text = if (i == last) (if (done) "You have finished this collection" else "Tap the circle to count") else ""
        }

        ring.setOnClickListener {
            if (count < items[i].target) {
                count++
                render()
                it.performHapticFeedback(
                    if (count == items[i].target) HapticFeedbackConstants.LONG_PRESS else HapticFeedbackConstants.VIRTUAL_KEY
                )
            }
        }

        (view as SwipeLayout).apply {
            card = view.findViewById(R.id.card)
            canPage = { dir -> if (dir > 0) i < last else i > 0 }
            onPage = { dir ->
                i += dir; count = 0
                arScroll.scrollTo(0, 0)
                render()
            }
        }
        render()
    }

    private fun immersive(on: Boolean) {
        val w = requireActivity().window
        val c = WindowCompat.getInsetsController(w, w.decorView)
        if (on) {
            c.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
            c.hide(WindowInsetsCompat.Type.systemBars())
        } else c.show(WindowInsetsCompat.Type.systemBars())
    }

    override fun onResume() { super.onResume(); immersive(true) }
    override fun onDestroyView() { immersive(false); super.onDestroyView() }

    companion object {
        fun of(key: String, start: Int = 0) = ReaderFragment().apply { arguments = bundleOf("key" to key, "start" to start) }
    }
}

/** 140dp ring: track #24504F, progress arc from 12 o'clock clockwise, 122dp inner disc #0D2124. */
class RingView @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : FrameLayout(ctx, attrs) {
    private var deg = 0f
    private var arcColor = context.getColor(R.color.gold)
    private val track = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = context.getColor(R.color.track) }
    private val arc = Paint(Paint.ANTI_ALIAS_FLAG)
    private val inner = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = context.getColor(R.color.bg) }

    init { setWillNotDraw(false) }

    fun setProgress(degrees: Float, color: Int) {
        deg = degrees; arcColor = color; invalidate()
    }

    override fun onDraw(canvas: Canvas) {
        val d = resources.displayMetrics.density
        val cx = width / 2f
        val cy = height / 2f
        val r = 70f * d
        canvas.drawCircle(cx, cy, r, track)
        if (deg > 0f) {
            arc.color = arcColor
            canvas.drawArc(RectF(cx - r, cy - r, cx + r, cy + r), -90f, deg, true, arc)
        }
        canvas.drawCircle(cx, cy, 61f * d, inner)
    }
}

/** Card-follows-finger swipe layout. dir: +1 = next (swipe left), -1 = previous (swipe right). */
class SwipeLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : LinearLayout(ctx, attrs) {
    var card: View? = null
    var canPage: ((Int) -> Boolean)? = null
    var onPage: ((Int) -> Unit)? = null

    private val vc = android.view.ViewConfiguration.get(ctx)
    private val slop = vc.scaledTouchSlop
    private val minFling = vc.scaledMinimumFlingVelocity * 8f
    private val dp = resources.displayMetrics.density
    private var downX = 0f
    private var downY = 0f
    private var dragging = false
    private var tapOk = false
    private var busy = false
    private var vt: android.view.VelocityTracker? = null

    /** Stops the system back gesture from stealing swipes that start at the screen edge. */
    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        super.onLayout(changed, l, t, r, b)
        if (android.os.Build.VERSION.SDK_INT >= 29) {
            val w = r - l
            val h = b - t
            val c = card
            val cy = if (c != null) (c.top + c.bottom) / 2 else h / 2
            val half = (100 * dp).toInt()
            val top = (cy - half).coerceAtLeast(0)
            val bottom = (cy + half).coerceAtMost(h)
            val edge = (40 * dp).toInt()
            setSystemGestureExclusionRects(
                listOf(android.graphics.Rect(0, top, edge, bottom), android.graphics.Rect(w - edge, top, w, bottom))
            )
        }
    }

    private fun inCard(x: Float, y: Float): Boolean {
        val c = card ?: return false
        return x >= c.left && x <= c.right && y >= c.top && y <= c.bottom
    }

    private fun far(e: MotionEvent) =
        kotlin.math.hypot(e.x - downX, e.y - downY) > slop

    private fun horizontal(e: MotionEvent) =
        kotlin.math.abs(e.x - downX) > slop && kotlin.math.abs(e.x - downX) > kotlin.math.abs(e.y - downY) * 1.2f

    private fun start(e: MotionEvent) {
        downX = e.x; downY = e.y; dragging = false
        tapOk = inCard(e.x, e.y)
        vt?.recycle(); vt = android.view.VelocityTracker.obtain(); vt?.addMovement(e)
    }

    private fun finishTap(e: MotionEvent) {
        if (tapOk && e.eventTime - e.downTime < android.view.ViewConfiguration.getLongPressTimeout()) next()
        tapOk = false
    }

    private fun cleanup() { vt?.recycle(); vt = null }

    override fun onInterceptTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
            }
            MotionEvent.ACTION_UP -> { if (!dragging) finishTap(e); cleanup() }
            MotionEvent.ACTION_CANCEL -> { tapOk = false; cleanup() }
        }
        return dragging
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        if (vt == null) vt = android.view.VelocityTracker.obtain()
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> start(e)
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && far(e)) tapOk = false
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
                if (dragging) drag(e.x - downX)
            }
            MotionEvent.ACTION_UP -> {
                vt?.addMovement(e)
                if (dragging) release(true) else finishTap(e)
                dragging = false; cleanup()
            }
            MotionEvent.ACTION_CANCEL -> {
                if (dragging) release(false)
                dragging = false; tapOk = false; cleanup()
            }
        }
        return true
    }

    private fun dirOf(t: Float) = if (t < 0f) 1 else -1   // finger moved left -> next

    private fun drag(dx: Float) {
        val c = card ?: return
        val d = dx - kotlin.math.sign(dx) * slop
        val ok = canPage?.invoke(dirOf(d)) ?: false
        val t = if (ok) d else d * 0.25f          // rubber band at the first/last azkar
        c.translationX = t
        c.alpha = 1f - 0.6f * minOf(kotlin.math.abs(t) / width.coerceAtLeast(1), 1f)
    }

    private fun release(up: Boolean) {
        val c = card ?: return
        vt?.computeCurrentVelocity(1000)
        val vx = vt?.xVelocity ?: 0f
        val t = c.translationX
        val dir = dirOf(t)
        val ok = canPage?.invoke(dir) ?: false
        val farEnough = kotlin.math.abs(t) > width * 0.28f
        val flung = kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(t) && kotlin.math.abs(t) > slop * 2
        if (up && ok && (farEnough || flung)) commit(c, dir)
        else c.animate().translationX(0f).alpha(1f).setDuration(200)
            .setInterpolator(android.view.animation.DecelerateInterpolator()).start()
    }

    /** Next zikr, or a small bounce if this is the last one (never leaves the reader). */
    fun next() {
        val c = card ?: return
        if (busy) return
        if (canPage?.invoke(1) == true) commit(c, 1)
        else c.animate().translationX(-12 * dp).setDuration(90)
            .withEndAction { c.animate().translationX(0f).setDuration(160).start() }.start()
    }

    private fun commit(c: View, dir: Int) {
        busy = true
        performHapticFeedback(android.view.HapticFeedbackConstants.VIRTUAL_KEY)
        val w = width.toFloat()
        c.animate().translationX(-dir * w).alpha(0f).setDuration(140)
            .setInterpolator(android.view.animation.AccelerateInterpolator())
            .withEndAction {
                onPage?.invoke(dir)
                c.translationX = dir * w * 0.35f   // new card enters from the side you swiped toward
                c.alpha = 0f
                c.animate().translationX(0f).alpha(1f).setDuration(220)
                    .setInterpolator(android.view.animation.DecelerateInterpolator())
                    .withEndAction { busy = false }.start()
            }.start()
    }
}

/** 52x32 track with 24dp knob, right-aligned in a 56x44 touch area. */
class ToggleSwitch @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : View(ctx, attrs) {
    var on = false
        set(v) { field = v; invalidate() }
    var onToggle: ((Boolean) -> Unit)? = null
    private val p = Paint(Paint.ANTI_ALIAS_FLAG)

    init {
        isClickable = true
        isFocusable = true
        setOnClickListener { on = !on; onToggle?.invoke(on) }
    }

    override fun onDraw(c: Canvas) {
        val d = resources.displayMetrics.density
        val left = width - 52f * d
        val top = (height - 32f * d) / 2f
        p.color = context.getColor(if (on) R.color.gold else R.color.track)
        c.drawRoundRect(RectF(left, top, left + 52f * d, top + 32f * d), 16f * d, 16f * d, p)
        p.color = context.getColor(if (on) R.color.bg else R.color.muted)
        val kx = left + (if (on) 24f else 4f) * d + 12f * d
        c.drawCircle(kx, top + 4f * d + 12f * d, 12f * d, p)
    }
}

/** Wrapping row layout with 8dp gaps (flex-wrap). */
class FlowLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : ViewGroup(ctx, attrs) {
    private val gap = (8 * resources.displayMetrics.density).toInt()

    override fun onMeasure(wSpec: Int, hSpec: Int) {
        val maxW = MeasureSpec.getSize(wSpec) - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            c.measure(MeasureSpec.makeMeasureSpec(maxW, MeasureSpec.AT_MOST), MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED))
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
        setMeasuredDimension(MeasureSpec.getSize(wSpec), y + lineH + paddingTop + paddingBottom)
    }

    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val maxW = r - l - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            c.layout(paddingLeft + x, paddingTop + y, paddingLeft + x + c.measuredWidth, paddingTop + y + c.measuredHeight)
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
    }
}

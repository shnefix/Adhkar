package com.azkar.app.ui

import android.os.Bundle
import android.text.Editable
import android.text.TextWatcher
import android.view.View
import android.widget.EditText
import android.widget.ImageView
import android.widget.LinearLayout
import android.widget.TextView
import androidx.fragment.app.Fragment
import com.azkar.app.R
import com.azkar.app.data.Builtins

class LibraryFragment : Fragment(R.layout.fragment_library) {

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val ctx = requireContext()
        bindNav(view, Tab.LIBRARY)
        val list = view.findViewById<LinearLayout>(R.id.list)
        val search = view.findViewById<EditText>(R.id.search)

        fun render(q: String) {
            list.removeAllViews()
            val query = q.trim().lowercase()
            val shown = Builtins.library.filter {
                query.isEmpty() || it.title.lowercase().contains(query) || it.libraryArabic.contains(query)
            }
            shown.forEachIndexed { i, c ->
                val v = layoutInflater.inflate(R.layout.item_library, list, false)
                v.findViewById<ImageView>(R.id.ic).setImageResource(c.icon)
                v.findViewById<TextView>(R.id.title).text = c.title
                v.findViewById<TextView>(R.id.sub_ar).text = c.libraryArabic
                v.setOnClickListener { nav().push(ReaderFragment.of(c.key)) }
                list.addView(v, Ui.lp(top = if (i > 0) 8 else 0, ctx = ctx))
            }
        }
        render("")
        search.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(s: CharSequence?, a: Int, b: Int, c: Int) {}
            override fun onTextChanged(s: CharSequence?, a: Int, b: Int, c: Int) {}
            override fun afterTextChanged(s: Editable?) = render(s?.toString() ?: "")
        })
    }
}

package com.azkar.app.ui

import android.os.Bundle
import android.view.View
import android.widget.EditText
import androidx.core.os.bundleOf
import androidx.fragment.app.Fragment
import com.azkar.app.R
import com.azkar.app.data.Store

class AddZikrFragment : Fragment(R.layout.fragment_add_zikr) {

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val newBox = view.findViewById<View>(R.id.newname_box)
        val newName = view.findViewById<EditText>(R.id.newname)
        val form = ZikrForm(view, allowNew = true) { newBox.visibility = if (it == "new") View.VISIBLE else View.GONE }
        form.n = 3
        form.sel = arguments?.getString("folder") ?: "new"

        view.findViewById<View>(R.id.btn_close).setOnClickListener { nav().pop() }
        view.findViewById<View>(R.id.btn_save).setOnClickListener {
            val text = form.arabic.text.toString().trim()
            if (text.isEmpty()) { form.arabic.requestFocus(); return@setOnClickListener }
            val folderId: String? = if (form.sel == "new") {
                val name = newName.text.toString().trim()
                if (name.isEmpty()) null
                else (Store.folders().firstOrNull { it.name.equals(name, ignoreCase = true) } ?: Store.addFolder(name)).id
            } else form.sel
            Store.addZikr(text, form.note.text.toString().trim(), form.n, folderId)
            nav().pop()
        }
    }

    companion object {
        /** [folder]: preselect this folder; null selects "+ New folder" like the original screen. */
        fun of(folder: String?) = AddZikrFragment().apply { arguments = bundleOf("folder" to folder) }
    }
}

package com.azkar.app.ui

import android.os.Bundle
import android.view.View
import androidx.core.os.bundleOf
import androidx.fragment.app.Fragment
import com.azkar.app.R
import com.azkar.app.data.Store

class EditZikrFragment : Fragment(R.layout.fragment_edit_zikr) {

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val z = Store.zikr(requireArguments().getString("id") ?: "")
        if (z == null) { nav().pop(); return }
        val form = ZikrForm(view, allowNew = false)
        form.arabic.setText(z.arabic)
        form.note.setText(z.note)
        form.n = z.repeat
        form.sel = z.folderId

        view.findViewById<View>(R.id.btn_close).setOnClickListener { nav().pop() }
        view.findViewById<View>(R.id.btn_save).setOnClickListener {
            val text = form.arabic.text.toString().trim()
            if (text.isEmpty()) { form.arabic.requestFocus(); return@setOnClickListener }
            Store.updateZikr(z.copy(arabic = text, note = form.note.text.toString().trim(), repeat = form.n, folderId = form.sel))
            nav().pop()
        }
        view.findViewById<View>(R.id.btn_delete).setOnClickListener {
            Store.deleteZikr(z.id)
            nav().pop()
        }
    }

    companion object {
        fun of(id: String) = EditZikrFragment().apply { arguments = bundleOf("id" to id) }
    }
}

package com.azkar.app.ui

import android.view.View
import android.widget.EditText
import android.widget.TextView
import com.azkar.app.R
import com.azkar.app.data.Builtins
import com.azkar.app.data.Store

/**
 * Shared behaviour of the Add zikr and Edit zikr screens: repeat stepper and folder chips.
 * [sel] is a folder key, "new" (Add only) or null.
 */
class ZikrForm(private val view: View, private val allowNew: Boolean, private val onSelChanged: (String?) -> Unit = {}) {
    var n = 3
        set(v) { field = v; renderStepper() }
    var sel: String? = null
        set(v) { field = v; renderChips(); onSelChanged(v) }

    val arabic: EditText = view.findViewById(R.id.ar)
    val note: EditText = view.findViewById(R.id.note)

    init {
        view.findViewById<View>(R.id.btn_dec).setOnClickListener { if (n > 1) n-- }
        view.findViewById<View>(R.id.btn_inc).setOnClickListener { n = minOf(999, n + 1) }
        renderStepper()
    }

    private fun renderStepper() {
        view.findViewById<TextView>(R.id.n).text = n.toString()
        view.findViewById<View>(R.id.btn_dec).alpha = if (n > 1) 1f else 0.4f
    }

    fun renderChips() {
        val ctx = view.context
        val box = view.findViewById<FlowLayout>(R.id.chips)
        box.removeAllViews()
        val choices = Builtins.folderChoices.map { it.key to it.title } + Store.folders().map { it.id to it.name }
        choices.forEach { (k, label) ->
            box.addView(Ui.chip(ctx, label, sel == k, false) { sel = k })
        }
        if (allowNew) box.addView(Ui.chip(ctx, "+ New folder", sel == "new", true) { sel = "new" })
    }
}

class ReaderFragment : Fragment(R.layout.fragment_reader) {

    private var count = 0

        super.onSaveInstanceState(outState)
        outState.putInt("i", i); outState.putInt("count", count)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {

        i = savedInstanceState?.getInt("i") ?: 0
        count = savedInstanceState?.getInt("count") ?: 0
        view.keepScreenOn = true

        val key = requireArguments().getString("key") ?: "morning"
        val col = Builtins.byKey(key)
        val items = Store.readerItems(key)
        if (items.isEmpty()) { nav().pop(); return }
        val title = when {
            key in listOf("morning", "evening", "afterPrayer", "sleep", "waking") -> (col?.title ?: "") + " azkar"
            col != null -> col.title
            else -> Store.folderName(key)
        }
        val sizeSp = intArrayOf(20, 26, 32)[Store.arabicSize().coerceIn(0, 2)]
        val last = items.size - 1
        val gold = ctx.getColor(R.color.gold)
        val teal = ctx.getColor(R.color.teal)
        val track = ctx.getColor(R.color.track)

        view.findViewById<View>(R.id.btn_back).setOnClickListener { nav().pop() }
        view.findViewById<TextView>(R.id.title).text = title
        val ar = view.findViewById<TextView>(R.id.ar).apply {
            textSize = sizeSp.toFloat()
            val lhPx = android.util.TypedValue.applyDimension(android.util.TypedValue.COMPLEX_UNIT_SP, sizeSp * 1.85f, resources.displayMetrics)
            androidx.core.widget.TextViewCompat.setLineHeight(this, lhPx.toInt())
        }
        val arScroll = view.findViewById<ScrollView>(R.id.ar_scroll)
        val segs = view.findViewById<LinearLayout>(R.id.segs)
        val ring = view.findViewById<RingView>(R.id.ring)
        val countTv = view.findViewById<TextView>(R.id.count)
        val sub = view.findViewById<TextView>(R.id.sub)
        val hint = view.findViewById<TextView>(R.id.hint)
        val pos = view.findViewById<TextView>(R.id.pos)

        fun render() {
            val cur = items[i]
            val done = count >= cur.target
            ar.text = cur.arabic
            pos.text = "${i + 1} of ${items.size}"
            segs.removeAllViews()
            items.indices.forEach { k ->
                val c = if (k < i) teal else if (k == i) gold else track
                val seg = View(ctx).apply { background = Ui.shape(ctx, c, null, 2f) }
                val p = LinearLayout.LayoutParams(0, Ui.pxi(ctx, 4f), 1f)
                if (k > 0) p.marginStart = Ui.pxi(ctx, 6f)
                segs.addView(seg, p)
            }
            countTv.text = count.toString()
            sub.text = if (done) "Done" else "of ${cur.target}"
            ring.setProgress(Math.round(count.toFloat() / cur.target * 360f).toFloat(), if (done) teal else gold)
            hint.text = if (i == last) (if (done) "You have finished this collection" else "Tap the circle to count") else ""
        }

        ring.setOnClickListener {
            if (count < items[i].target) {
                count++
                render()
                it.performHapticFeedback(
                    if (count == items[i].target) HapticFeedbackConstants.LONG_PRESS else HapticFeedbackConstants.VIRTUAL_KEY
                )
            }
        }

        view.findViewById<SwipeLayout>(R.id.swipe).apply {
            canPage = { dir -> if (dir > 0) i < last else i > 0 }

                render()
            }
        }
        render()
    }

            c.hide(WindowInsetsCompat.Type.systemBars())
        } else c.show(WindowInsetsCompat.Type.systemBars())
    }

    override fun onResume() { super.onResume(); immersive(true) }
    override fun onDestroyView() { immersive(false); super.onDestroyView() }

    companion object {
        fun of(key: String) = ReaderFragment().apply { arguments = bundleOf("key" to key) }
    }
}

/** Card-follows-finger swipe layout. dir: +1 = next (swipe left), -1 = previous (swipe right). */
class SwipeLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : LinearLayout(ctx, attrs) {
    var canPage: ((Int) -> Boolean)? = null
    var onPage: ((Int) -> Unit)? = null

    private val vc = android.view.ViewConfiguration.get(ctx)
    private val slop = vc.scaledTouchSlop
    private val minFling = vc.scaledMinimumFlingVelocity * 8f
    private var downX = 0f
    private var downY = 0f
    private var dragging = false
    private var busy = false
    private var vt: android.view.VelocityTracker? = null
    private val card: View? get() = getChildAt(0)

    /** Stops the system back gesture from stealing swipes that start at the screen edge. */
    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        super.onLayout(changed, l, t, r, b)
        if (android.os.Build.VERSION.SDK_INT >= 29) {
            systemGestureExclusionRects = listOf(android.graphics.Rect(0, 0, r - l, b - t))
        }
    }

    private fun horizontal(e: MotionEvent) =
        kotlin.math.abs(e.x - downX) > slop && kotlin.math.abs(e.x - downX) > kotlin.math.abs(e.y - downY) * 1.2f

    override fun onInterceptTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                downX = e.x; downY = e.y; dragging = false
                vt?.recycle(); vt = android.view.VelocityTracker.obtain(); vt?.addMovement(e)
            }
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> { vt?.recycle(); vt = null }
        }
        return dragging
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        if (vt == null) vt = android.view.VelocityTracker.obtain()
        vt?.addMovement(e)
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> { downX = e.x; downY = e.y; return true }
            MotionEvent.ACTION_MOVE -> {
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
                if (dragging) drag(e.x - downX)
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (dragging) release(e.actionMasked == MotionEvent.ACTION_UP)
                dragging = false
                vt?.recycle(); vt = null
            }
        }
        return true
    }

    private fun dirOf(t: Float) = if (t < 0f) 1 else -1   // finger moved left -> next

    private fun drag(dx: Float) {
        val c = card ?: return
        val d = dx - kotlin.math.sign(dx) * slop
        val ok = canPage?.invoke(dirOf(d)) ?: false
        val t = if (ok) d else d * 0.25f          // rubber band at the first/last azkar
        c.translationX = t
        c.alpha = 1f - 0.6f * minOf(kotlin.math.abs(t) / width.coerceAtLeast(1), 1f)
    }

    private fun release(up: Boolean) {
        val c = card ?: return
        vt?.computeCurrentVelocity(1000)
        val vx = vt?.xVelocity ?: 0f
        val t = c.translationX
        val dir = dirOf(t)
        val ok = canPage?.invoke(dir) ?: false
        val far = kotlin.math.abs(t) > width * 0.28f
        val flung = kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(t) && kotlin.math.abs(t) > slop * 2
        if (up && ok && (far || flung)) commit(c, dir)
        else c.animate().translationX(0f).alpha(1f).setDuration(200)
            .setInterpolator(android.view.animation.DecelerateInterpolator()).start()
    }

    private fun commit(c: View, dir: Int) {
        busy = true
        performHapticFeedback(android.view.HapticFeedbackConstants.VIRTUAL_KEY)
        val w = width.toFloat()
        c.animate().translationX(-dir * w).alpha(0f).setDuration(140)
            .setInterpolator(android.view.animation.AccelerateInterpolator())
            .withEndAction {
                onPage?.invoke(dir)
                c.translationX = dir * w * 0.35f   // new card enters from the side you swiped toward
                c.alpha = 0f
                c.animate().translationX(0f).alpha(1f).setDuration(220)
                    .setInterpolator(android.view.animation.DecelerateInterpolator())
                    .withEndAction { busy = false }.start()
            }.start()
    }
}

/** Wrapping row layout with 8dp gaps (flex-wrap). */
class FlowLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : ViewGroup(ctx, attrs) {
    private val gap = (8 * resources.displayMetrics.density).toInt()

    override fun onMeasure(wSpec: Int, hSpec: Int) {
        val maxW = MeasureSpec.getSize(wSpec) - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            c.measure(MeasureSpec.makeMeasureSpec(maxW, MeasureSpec.AT_MOST), MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED))
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
        setMeasuredDimension(MeasureSpec.getSize(wSpec), y + lineH + paddingTop + paddingBottom)
    }

    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val maxW = r - l - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            c.layout(paddingLeft + x, paddingTop + y, paddingLeft + x + c.measuredWidth, paddingTop + y + c.measuredHeight)
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
    }
}

# Azkar (native Kotlin)

Native Android (Kotlin + XML) conversion of the nine HTML screens. No WebView, no HTML/JS at runtime.

Build: open this folder in Android Studio (Koala or newer, JDK 17), let Gradle sync, then
**Build > Build APK(s)** (or Run on a device). minSdk 26, targetSdk 34. Package: `com.azkar.app`.

## Screens
Home, Library, My azkar, Reader, Add zikr, Edit zikr, Edit folder (also the folder view), Customize Home, Reminders.

## Where the design values came from
Every colour, size, radius, padding, gap, font size/weight and line height is copied from the inline CSS of the
screens (CSS px = dp, font px = sp). Icons are the original SVG paths as vector drawables. Fonts are the DM Sans
(400/500/700) and Amiri (400/700) files embedded in the HTML, converted to TTF in `res/font`.
Placeholder text uses Chrome's default placeholder grey (#757575) because the HTML does not set one.

## Storage
`data/Store.kt` keeps azkar, folders, Home customisation, reader settings and reminder switches in
SharedPreferences (JSON). Everything persists across restarts.

## Reminders
Daily local notifications at 6:00 AM (Morning), 5:00 PM (Evening), 10:00 PM (Before sleep), using inexact alarms
(no special permission), re-armed after reboot. Android 13+ asks for notification permission when a reminder is switched on.

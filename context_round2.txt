package com.azkar.app

import android.Manifest
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Build
import android.os.Bundle
import android.view.WindowManager
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.WindowCompat
import androidx.fragment.app.Fragment
import androidx.fragment.app.FragmentManager
import com.azkar.app.data.Store
import com.azkar.app.ui.HomeFragment
import com.azkar.app.ui.LibraryFragment
import com.azkar.app.ui.MyAzkarFragment
import com.azkar.app.ui.ReaderFragment
import com.azkar.app.ui.RemindersFragment
import com.azkar.app.ui.Tab

class MainActivity : AppCompatActivity() {
    private var currentTab: Tab = Tab.HOME
    private val notifPermission = registerForActivityResult(ActivityResultContracts.RequestPermission()) { }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        Store.init(this)
        if (Build.VERSION.SDK_INT >= 28) {
            window.attributes = window.attributes.apply {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
        }
        if (Build.VERSION.SDK_INT >= 33 &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) notifPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
        WindowCompat.setDecorFitsSystemWindows(window, false)
        WindowCompat.getInsetsController(window, window.decorView).apply {
            isAppearanceLightStatusBars = false
            isAppearanceLightNavigationBars = false
        }
        setContentView(R.layout.activity_main)
        if (savedInstanceState == null) {
            showRoot(HomeFragment())
            handleIntent(intent)
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        handleIntent(intent)
    }

    /** A reminder notification opens its collection in the reader. */
    private fun handleIntent(i: Intent?) {
        val key = i?.getStringExtra("open") ?: return
        i.removeExtra("open")
        showRoot(HomeFragment())
        push(ReaderFragment.of(key))
    }

    private fun showRoot(f: Fragment) {
        supportFragmentManager.popBackStackImmediate(null, FragmentManager.POP_BACK_STACK_INCLUSIVE)
        supportFragmentManager.beginTransaction().replace(R.id.container, f).commit()
    }

    fun openTab(tab: Tab) {
        if (tab == currentTab && supportFragmentManager.backStackEntryCount == 0) return
        currentTab = tab
        showRoot(
            when (tab) {
                Tab.HOME -> HomeFragment()
                Tab.LIBRARY -> LibraryFragment()
                Tab.MY -> MyAzkarFragment()
                Tab.REMINDERS -> RemindersFragment()
            }
        )
    }

    fun push(f: Fragment) {
        supportFragmentManager.beginTransaction().replace(R.id.container, f).addToBackStack(null).commit()
    }

    fun pop() {
        supportFragmentManager.popBackStack()
    }
}

            override fun afterTextChanged(s: Editable?) {

            Store.setHidden(if (on) Store.hidden() - id else Store.hidden() + id)
            renderHome()
        }

        val empty = view.findViewById<View>(R.id.empty)
        val label = view.findViewById<TextView>(R.id.order_label)

        fun renderRows() {
            val list = Store.zikrsIn(id)
            label.text = "Order · " + list.size + if (list.size == 1) " zikr" else " azkar"
            empty.visibility = if (list.isEmpty()) View.VISIBLE else View.GONE
            rows.removeAllViews()
            val last = list.size - 1
            list.forEachIndexed { idx, z ->
                val v = layoutInflater.inflate(R.layout.item_folder_zikr, rows, false)
                v.findViewById<TextView>(R.id.ar).text = z.arabic
                v.findViewById<TextView>(R.id.line).text = "×${z.repeat}" + if (z.note.isBlank()) "" else " · ${z.note}"

                val down = v.findViewById<View>(R.id.down)
                up.alpha = if (idx == 0) 0.35f else 1f
                down.alpha = if (idx == last) 0.35f else 1f
                fun move(d: Int) {
                    val ids = list.map { it.id }.toMutableList()
                    val t = ids[idx]; ids[idx] = ids[idx + d]; ids[idx + d] = t
                    Store.reorderFolder(id, ids)
                    renderRows()

                up.setOnClickListener { if (idx > 0) move(-1) }

            }

        renderRows()

        view.findViewById<View>(R.id.btn_add).setOnClickListener { nav().push(AddZikrFragment.of(id)) }

            if (cards.isEmpty()) return null
            val box = LinearLayout(ctx).apply { orientation = LinearLayout.VERTICAL }
            if (label != null) box.addView(Ui.caps(ctx, label))
            box.addView(Ui.grid(ctx, cards, 2, 12), Ui.lp(top = if (label != null) 12 else 0, ctx = ctx))
            return box
        }

        val daily = section("Daily collections", collections.map { collectionCard(it) })
        val mine = section(null, folders.map { folderCard(it) })

        val parts: List<View?> = when (Store.placement()) {
            Placement.TOP -> listOf(mine, daily)
            Placement.BOTTOM -> listOf(daily, mine)
            Placement.MIXED -> listOf(
                section("Daily collections", order.mapNotNull { k ->
                    Builtins.homeCollections.firstOrNull { it.key == k }?.let { collectionCard(it) }
                        ?: Store.folder(k)?.let { folderCard(it) }
                })
            )
        }
        parts.filterNotNull().forEachIndexed { i, v -> sections.addView(v, Ui.lp(top = if (i == 0) 16 else 20, ctx = ctx)) }
    }

    private var count = 0

        outState.putInt("i", i); outState.putInt("count", count)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        val ctx = requireContext()
        i = savedInstanceState?.getInt("i") ?: 0
        count = savedInstanceState?.getInt("count") ?: 0
        view.keepScreenOn = true

        val key = requireArguments().getString("key") ?: "morning"
        val col = Builtins.byKey(key)
        val items = Store.readerItems(key)
        if (items.isEmpty()) { nav().pop(); return }
        val title = when (key) {
            "morning", "evening", "afterPrayer", "sleep", "waking" -> (col?.title ?: "") + " azkar"
            else -> col?.title ?: ""
        }
        val sizeSp = intArrayOf(20, 26, 32)[Store.arabicSize().coerceIn(0, 2)]
        val last = items.size - 1
        val gold = ctx.getColor(R.color.gold)
        val teal = ctx.getColor(R.color.teal)
        val track = ctx.getColor(R.color.track)

        view.findViewById<View>(R.id.btn_back).setOnClickListener { nav().pop() }
        view.findViewById<TextView>(R.id.title).text = title
        val ar = view.findViewById<TextView>(R.id.ar).apply {
            textSize = sizeSp.toFloat()
            val lhPx = android.util.TypedValue.applyDimension(android.util.TypedValue.COMPLEX_UNIT_SP, sizeSp * 1.85f, resources.displayMetrics)
            androidx.core.widget.TextViewCompat.setLineHeight(this, lhPx.toInt())
        }
        val arScroll = view.findViewById<ScrollView>(R.id.ar_scroll)
        val segs = view.findViewById<LinearLayout>(R.id.segs)
        val ring = view.findViewById<RingView>(R.id.ring)
        val countTv = view.findViewById<TextView>(R.id.count)
        val sub = view.findViewById<TextView>(R.id.sub)
        val hint = view.findViewById<TextView>(R.id.hint)
        val pos = view.findViewById<TextView>(R.id.pos)

        fun render() {
            val cur = items[i]
            val done = count >= cur.target
            ar.text = cur.arabic
            pos.text = "${i + 1} of ${items.size}"
            segs.removeAllViews()
            items.indices.forEach { k ->
                val c = if (k < i) teal else if (k == i) gold else track
                val seg = View(ctx).apply { background = Ui.shape(ctx, c, null, 2f) }
                val p = LinearLayout.LayoutParams(0, Ui.pxi(ctx, 4f), 1f)
                if (k > 0) p.marginStart = Ui.pxi(ctx, 6f)
                segs.addView(seg, p)
            }
            countTv.text = count.toString()
            sub.text = if (done) "Done" else "of ${cur.target}"
            ring.setProgress(Math.round(count.toFloat() / cur.target * 360f).toFloat(), if (done) teal else gold)
            hint.text = if (i == last) (if (done) "You have finished this collection" else "Tap the circle to count")
            else "Tap to count · swipe right for next"
        }

        ring.setOnClickListener {
            if (count < items[i].target) {
                count++
                render()
                it.performHapticFeedback(
                    if (count == items[i].target) HapticFeedbackConstants.LONG_PRESS else HapticFeedbackConstants.VIRTUAL_KEY
                )
            }
        }

        view.findViewById<SwipeLayout>(R.id.swipe).apply {
            canPage = { dir -> if (dir > 0) i < last else i > 0 }
            onPage = { dir ->
                i += dir; count = 0
                arScroll.scrollTo(0, 0)
                render()
            }
        }
        render()
    }

    private fun immersive(on: Boolean) {
        val w = requireActivity().window
        val c = WindowCompat.getInsetsController(w, w.decorView)
        if (on) {
            c.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
            c.hide(WindowInsetsCompat.Type.systemBars())
        } else c.show(WindowInsetsCompat.Type.systemBars())
    }

    override fun onDestroyView() { immersive(false); super.onDestroyView() }

    }
}

/** 140dp ring: track #24504F, progress arc from 12 o'clock clockwise, 122dp inner disc #0D2124. */
class RingView @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : FrameLayout(ctx, attrs) {
    private var deg = 0f
    private var arcColor = context.getColor(R.color.gold)
    private val track = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = context.getColor(R.color.track) }
    private val arc = Paint(Paint.ANTI_ALIAS_FLAG)
    private val inner = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = context.getColor(R.color.bg) }

    init { setWillNotDraw(false) }

    fun setProgress(degrees: Float, color: Int) {
        deg = degrees; arcColor = color; invalidate()
    }

    override fun onDraw(canvas: Canvas) {
        val d = resources.displayMetrics.density
        val cx = width / 2f
        val cy = height / 2f
        val r = 70f * d
        canvas.drawCircle(cx, cy, r, track)
        if (deg > 0f) {
            arc.color = arcColor
            canvas.drawArc(RectF(cx - r, cy - r, cx + r, cy + r), -90f, deg, true, arc)
        }
        canvas.drawCircle(cx, cy, 61f * d, inner)
    }
}

/** Card-follows-finger swipe layout. */
class SwipeLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : LinearLayout(ctx, attrs) {
    /** dir: +1 = next (swipe right), -1 = previous (swipe left). */
    var canPage: ((Int) -> Boolean)? = null
    var onPage: ((Int) -> Unit)? = null

    private val vc = android.view.ViewConfiguration.get(ctx)
    private val slop = vc.scaledTouchSlop
    private val minFling = vc.scaledMinimumFlingVelocity * 8f
    private var downX = 0f
    private var downY = 0f
    private var dragging = false
    private var busy = false
    private var vt: android.view.VelocityTracker? = null
    private val card: View? get() = getChildAt(0)

    private fun horizontal(e: MotionEvent) =
        kotlin.math.abs(e.x - downX) > slop && kotlin.math.abs(e.x - downX) > kotlin.math.abs(e.y - downY) * 1.2f

    override fun onInterceptTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                downX = e.x; downY = e.y; dragging = false
                vt?.recycle(); vt = android.view.VelocityTracker.obtain(); vt?.addMovement(e)
            }
            MotionEvent.ACTION_MOVE -> {
                vt?.addMovement(e)
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> { vt?.recycle(); vt = null }
        }
        return dragging
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (busy) return true
        if (vt == null) vt = android.view.VelocityTracker.obtain()
        vt?.addMovement(e)
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> { downX = e.x; downY = e.y; return true }
            MotionEvent.ACTION_MOVE -> {
                if (!dragging && horizontal(e)) { dragging = true; parent?.requestDisallowInterceptTouchEvent(true) }
                if (dragging) drag(e.x - downX)
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (dragging) release(e.actionMasked == MotionEvent.ACTION_UP)
                dragging = false
                vt?.recycle(); vt = null
            }
        }
        return true
    }

    private fun drag(dx: Float) {
        val c = card ?: return
        val d = dx - kotlin.math.sign(dx) * slop
        val ok = canPage?.invoke(if (d > 0) 1 else -1) ?: false
        val t = if (ok) d else d * 0.25f
        c.translationX = t
        c.alpha = 1f - 0.6f * minOf(kotlin.math.abs(t) / width.coerceAtLeast(1), 1f)
    }

    private fun release(up: Boolean) {
        val c = card ?: return
        vt?.computeCurrentVelocity(1000)
        val vx = vt?.xVelocity ?: 0f
        val t = c.translationX
        val dir = if (t > 0) 1 else -1
        val ok = canPage?.invoke(dir) ?: false
        val far = kotlin.math.abs(t) > width * 0.28f
        val flung = kotlin.math.abs(vx) > minFling && kotlin.math.sign(vx) == kotlin.math.sign(t) && kotlin.math.abs(t) > slop * 2
        if (up && ok && (far || flung)) commit(c, dir)
        else c.animate().translationX(0f).alpha(1f).setDuration(200)
            .setInterpolator(android.view.animation.DecelerateInterpolator()).start()
    }

    private fun commit(c: View, dir: Int) {
        busy = true
        performHapticFeedback(android.view.HapticFeedbackConstants.VIRTUAL_KEY)
        val w = width.toFloat()
        c.animate().translationX(dir * w).alpha(0f).setDuration(140)
            .setInterpolator(android.view.animation.AccelerateInterpolator())
            .withEndAction {
                onPage?.invoke(dir)
                c.translationX = -dir * w * 0.35f
                c.alpha = 0f
                c.animate().translationX(0f).alpha(1f).setDuration(220)
                    .setInterpolator(android.view.animation.DecelerateInterpolator())
                    .withEndAction { busy = false }.start()
            }.start()
    }
}

/** 52x32 track with 24dp knob, right-aligned in a 56x44 touch area. */
class ToggleSwitch @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : View(ctx, attrs) {
    var on = false
        set(v) { field = v; invalidate() }
    var onToggle: ((Boolean) -> Unit)? = null
    private val p = Paint(Paint.ANTI_ALIAS_FLAG)

    init {
        isClickable = true
        isFocusable = true
        setOnClickListener { on = !on; onToggle?.invoke(on) }
    }

    override fun onDraw(c: Canvas) {
        val d = resources.displayMetrics.density
        val left = width - 52f * d
        val top = (height - 32f * d) / 2f
        p.color = context.getColor(if (on) R.color.gold else R.color.track)
        c.drawRoundRect(RectF(left, top, left + 52f * d, top + 32f * d), 16f * d, 16f * d, p)
        p.color = context.getColor(if (on) R.color.bg else R.color.muted)

/** Wrapping row layout with 8dp gaps (flex-wrap). */
class FlowLayout @JvmOverloads constructor(ctx: Context, attrs: AttributeSet? = null) : ViewGroup(ctx, attrs) {
    private val gap = (8 * resources.displayMetrics.density).toInt()

    override fun onMeasure(wSpec: Int, hSpec: Int) {
        val maxW = MeasureSpec.getSize(wSpec) - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            c.measure(MeasureSpec.makeMeasureSpec(maxW, MeasureSpec.AT_MOST), MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED))
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
        setMeasuredDimension(MeasureSpec.getSize(wSpec), y + lineH + paddingTop + paddingBottom)
    }

    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val maxW = r - l - paddingLeft - paddingRight
        var x = 0; var y = 0; var lineH = 0
        for (i in 0 until childCount) {
            val c = getChildAt(i)
            if (c.visibility == GONE) continue
            if (x > 0 && x + c.measuredWidth > maxW) { x = 0; y += lineH + gap; lineH = 0 }
            c.layout(paddingLeft + x, paddingTop + y, paddingLeft + x + c.measuredWidth, paddingTop + y + c.measuredHeight)
            x += c.measuredWidth + gap
            lineH = maxOf(lineH, c.measuredHeight)
        }
    }
}
